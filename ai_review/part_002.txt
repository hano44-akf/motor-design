"""Explicit nominal/capacity execution plan and saved evidence coverage."""
from collections import Counter


def condition_plan(request):
    op=request['operating_point'];margin=op.get('torque_margin_pct',0)
    return [dict(role=role,temperature_C=temp,voltage_V=bus,
                 target_torque_Nm=op['torque_Nm']*factor,
                 condition_id=f'{role}_t{ti}_v{vi}',temperature_index=ti,voltage_index=vi)
            for ti,temp in enumerate(request['temperatures_C'])
            for vi,bus in enumerate(op['voltages_V'])
            for role,factor in ([('nominal',1),('capacity',1+margin/100)] if margin>0 else [('nominal',1)])]


def run_key(run):
    return (run.get('verification_role','nominal'),run.get('input_options',{}).get('temperature_C'),
            run.get('design',{}).get('voltage_V'))


def coverage(runs,request):
    expected=Counter((p['role'],p['temperature_C'],p['voltage_V']) for p in condition_plan(request))
    actual=Counter(run_key(r) for r in runs)
    return {'complete':actual==expected,'missing':[list(k) for k in (expected-actual).elements()],
            'unexpected_or_duplicate':[list(k) for k in (actual-expected).elements()]}


def capacity_partner(run,runs,request):
    if run_key(run)[0]!='nominal' or request['operating_point'].get('torque_margin_pct',0)<=0:return None
    key=('capacity',*run_key(run)[1:]);matches=[r for r in runs if run_key(r)==key]
    if len(matches)!=1:return None
    partner=matches[0]
    fields=('diameter_mm','length_mm','turns_per_coil','wire_diameter_mm','airgap_mm','geometry_mm')
    if any(run.get('design',{}).get(k)!=partner.get('design',{}).get(k) for k in fields):return None
    if run.get('geometry_mm')!=partner.get('geometry_mm'):return None
    return partner

def normalize_request(request):
    """Fill explicit compatibility defaults without mutating the caller's input."""
    if not isinstance(request,dict):raise ValueError('Request must be an object')
    result=copy.deepcopy(request)
    result.setdefault('temperatures_C',[20])
    result.setdefault('bounds',[[55,100],[20,65],[1,8],[.8,2.2]])
    options=result.setdefault('spm',{})
    if not isinstance(options,dict):raise ValueError('spm must be an object')
    rotor=resolve_rotor(options)
    options.setdefault('shape',rotor['input_value'])
    options.setdefault('magnetisation',default_magnetisation(options))
    op=result.get('operating_point')
    if isinstance(op,dict):
        op.setdefault('torque_margin_pct',0)
        voltages=op.get('voltages_V')
        if isinstance(voltages,list) and voltages and all(isinstance(v,(int,float)) and not isinstance(v,bool) and math.isfinite(v) for v in voltages):
            op['voltages_V']=sorted(set(voltages))
    return result


def number(value,name,minimum=None,maximum=None,strict=False):
    if isinstance(value,bool) or not isinstance(value,(int,float)) or not math.isfinite(value):
        raise ValueError('Finite numeric value required: '+name)
    if minimum is not None and (value<=minimum if strict else value<minimum):raise ValueError('Value below allowed domain: '+name)
    if maximum is not None and value>maximum:raise ValueError('Value above allowed domain: '+name)
    return value


def validate_input(request):
    """Validate syntax/physical domains, independently of available calibration."""
    if not isinstance(request,dict) or request.get('schema')!='spm-study-v1':raise ValueError('Expected spm-study-v1 request')
    for key in ('core','magnet'):
        if not isinstance(request.get(key),str) or not request[key].strip():raise ValueError('Material name required: '+key)
    if request.get('connection') not in ('delta','star'):raise ValueError('Connection must be delta or star')
    for key in ('strands','paths'):
        value=number(request.get(key),key,1)
        if int(value)!=value:raise ValueError('Positive integer required: '+key)
    if request.get('control') not in ('FOC_Id0_SVPWM','FOC_MTPA_SVPWM','six_step'):
        raise ValueError('Unknown control specification')
    if request.get('dimension_resolution_mm',.1)!=.1:raise ValueError('Design dimension resolution is fixed at 0.1 mm')
    for key in ('detailed_review','open_motorcad_after'):
        if not isinstance(request.get(key,False),bool):raise ValueError(key+' must be boolean')
    op=request.get('operating_point')
    if not isinstance(op,dict):raise ValueError('Operating point required')
    for key in ('torque_Nm','speed_rpm'):number(op.get(key),key,0,strict=True)
    number(op.get('torque_margin_pct',0),'torque_margin_pct',0)
    eta=op.get('efficiency_min_pct')
    if eta is not None:number(eta,'efficiency_min_pct',0,100)
    for key in ('current_density_max_A_mm2','dc_current_max_A'):
        if op.get(key) is not None:number(op[key],key,0)
    voltages=op.get('voltages_V')
    if not isinstance(voltages,list) or not voltages:raise ValueError('DC voltage conditions required')
    for value in voltages:number(value,'voltages_V',0,strict=True)
    nominal=op.get('nominal_voltage_V')
    if nominal is not None:
        number(nominal,'nominal_voltage_V',0,strict=True)
        # A saved-design verification may evaluate only a subset of buses while
        # retaining the original design's nominal voltage as metadata.
        if not request.get('formula_source_job') and not min(voltages)<=nominal<=max(voltages):raise ValueError('Nominal voltage must lie within voltage conditions')
    temps=request.get('temperatures_C',[20])
    if not isinstance(temps,list) or not temps:raise ValueError('Temperature conditions required')
    for value in temps:number(value,'temperatures_C',-273.15,strict=True)
    bounds=request.get('bounds',[[55,100],[20,65],[1,8],[.8,2.2]])
    if not isinstance(bounds,list) or len(bounds)!=4:raise ValueError('Four design bounds required')
    for i,bound in enumerate(bounds):
        if not isinstance(bound,(tuple,list)) or len(bound)!=2:raise ValueError('Invalid bounds')
        lo,hi=[number(v,'bounds',0,strict=True) for v in bound]
        if lo>hi:raise ValueError('Reversed bounds')
        factor=1 if i==2 else 10
        if i==2 and any(not float(v).is_integer() for v in bound):raise ValueError('Turns bounds must be integers')

    if value is None or value=='':return 10000.0

        raise ValueError('Switching frequency must be positive Hz, or blank for 10000 Hz')

def pole_limit(speed_rpm,frequency_hz=None,poles=None):
    frequency=switching_hz(frequency_hz)
    if isinstance(speed_rpm,bool) or not isinstance(speed_rpm,(int,float)) or not math.isfinite(speed_rpm) or speed_rpm<=0:
        raise ValueError('Positive speed required for pole limit')
    # Compute in pole pairs so the cap is always even. Equality at 20 is allowed.
    max_pairs=math.floor(math.nextafter(60*frequency/(20*speed_rpm),math.inf))
    result={'switching_frequency_hz':frequency,'minimum_switching_cycles':20,
            'max_even_poles':2*max_pairs,'speed_rpm':speed_rpm,
            'definition':'PWM carrier periods per fundamental electrical period; not transistor edge count'}
    if poles is not None:
        if isinstance(poles,bool) or not isinstance(poles,(int,float)) or not math.isfinite(poles) or poles<2 or poles%2:
            raise ValueError('Pole count must be an even integer >= 2')
        electrical_hz=poles*speed_rpm/120
        result.update(poles=int(poles),electrical_frequency_hz=electrical_hz,
                      switching_cycles_per_electrical_cycle=frequency/electrical_hz,
                      passes=int(poles)<=result['max_even_poles'])
    return result

def inductance_voltage_upper(run, voltage_margin=VOLTAGE_RESERVE_FACTOR):
    """FOC Id=0, SVPWM fundamental winding-phase RMS voltage budget, SI output.

    if abs(readback.get('PhaseAdvance',0))>1e-7:

            'scope':'FOC Id=0/SVPWM, operating-point Lq voltage budget only; not differential/HF inductance or delta circulating-current validation'}

"""Common iron flux-density ceiling for Motor-CAD path maximum outputs (tesla)."""
import math

IRON_FIELDS = (
    'Stator Tooth Flux Density (maximum)',
    'Stator Tooth Tip Flux Density (maximum)',
    'Stator Back Iron Flux Density (maximum)',
    'Rotor Back Iron Flux Density (maximum)',
)


def flux_density_limit(value=None):
    if value is None or value == '':
        return 1.8
    if isinstance(value, bool) or not isinstance(value, (int, float)) or not math.isfinite(value) or value <= 0:
        raise ValueError('Maximum iron flux density must be positive tesla; blank defaults to 1.8 T')
    return float(value)


def flux_density_review(run, value=None):
    limit = flux_density_limit(value)
    regions = {}
    for field in IRON_FIELDS:
        actual = run['values'].get(field)
        usable = isinstance(actual, (int, float)) and not isinstance(actual, bool) and math.isfinite(actual) and actual >= 0
        regions[field] = {'actual_T': actual if usable else None,
                          'passed': actual <= limit if usable else None}
    missing = [k for k, r in regions.items() if r['actual_T'] is None]
    maximum = None if missing else max(r['actual_T'] for r in regions.values())
    passed = False if any(r['passed'] is False for r in regions.values()) else None if missing else True
    return {'limit_T': limit, 'maximum_T': maximum, 'regions': regions,
            'missing_regions': missing, 'passed': passed,
            'status': 'incomplete' if missing else 'pass' if passed else 'exceeded',
            'basis': 'Motor-CAD reported iron path maxima; not a global mesh-element maximum or saturation percentage'}
