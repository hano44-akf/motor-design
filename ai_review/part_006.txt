def digest(data):
    return hashlib.sha256(data).hexdigest()

def parse_mdb(data):
    for encoding in ('utf-8-sig', 'cp1252', 'cp949'):
        try:
            text = data.decode(encoding)
            break
        except UnicodeDecodeError:
            continue
    else:
        raise ValueError('Unsupported encoding; original bytes retained')
    sections, current, preamble = [], None, []
    for number, line in enumerate(text.splitlines(), 1):
        stripped = line.strip()
        if stripped.startswith('[') and stripped.endswith(']'):
            current = {'name': stripped[1:-1], 'raw_header': line, 'line': number, 'entries': [], 'other_lines': []}
            sections.append(current)
        elif current is None:
            preamble.append({'line': number, 'text': line})
        elif '=' in line:
            key, value = line.split('=', 1)
            current['entries'].append({'key': key, 'value': value})
        else:
            current['other_lines'].append({'line': number, 'text': line})
    return encoding, sections, preamble

def summarize(section):
    values = {}
    warnings = []
    arrays = {}
    for entry in section['entries']:
        key, value = entry['key'], entry['value']
        if key in values:
            warnings.append('duplicate_key:' + key)
        values[key] = value
        match = re.fullmatch(r'(.+)\[(\d+)\]', key)
        if match:
            arrays.setdefault(match[1], []).append({'index': int(match[2]), 'raw_value': value})
    for prefix, points in arrays.items():
        points.sort(key=lambda p: p['index'])
        indices = [p['index'] for p in points]
        if indices != list(range(len(indices))):
            warnings.append('non_contiguous_or_duplicate_indices:' + prefix)
    curves = []
    for kind, keys in [('steel_bh', ('BValue', 'HValue')),
                       ('magnet_bh', ('Temperature', 'BValue_Magnet', 'HValue_Magnet')),
                       ('steel_loss', ('Frequency', 'FluxDensity', 'LossDensity'))]:
        if not any(k in arrays for k in keys):
            continue
        indices = [{p['index'] for p in arrays.get(k, [])} for k in keys]
        complete = all(i == indices[0] for i in indices)
        if not complete:
            warnings.append('mismatched_curve_indices:' + kind)
        curves.append({'kind': kind, 'fields': list(keys), 'point_count': len(set.intersection(*indices)),
                       'indices_match': complete, 'units_status': 'raw_source_fields_not_normalized'})
    return values, arrays, curves, warnings

def write_csv(path, rows, fields):
    with path.open('w', newline='', encoding='utf-8-sig') as stream:
        writer = csv.DictWriter(stream, fieldnames=fields)
        writer.writeheader()
        writer.writerows(rows)

def convert(source):
    data = source.read_bytes()
    text = data.decode('utf-8-sig')
    fields = {}
    name = None
    for line in text.splitlines():
        if line.startswith('[') and line.endswith(']'):
            if name is not None:
                raise ValueError('Expected exactly one material section')
            name = line[1:-1]
        elif '=' in line:
            key, value = line.split('=', 1)
            if key in fields:
                raise ValueError('Duplicate field: ' + key)
            fields[key] = value
        elif line.strip():
            raise ValueError('Unrecognized line: ' + line)
    if fields.get('Solid Type') != 'Magnet':
        raise ValueError('Only single-material magnet MDB supported')
    arrays = {}
    for prefix in ('Temperature', 'BValue_Magnet', 'HValue_Magnet'):
        arrays[prefix] = {int(m.group(1)): float(value)
                          for key, value in fields.items()
                          if (m := re.fullmatch(re.escape(prefix) + r'\[(\d+)\]', key))}
    indices = set(arrays['Temperature'])
    if not indices or any(set(a) != indices for a in arrays.values()):
        raise ValueError('Missing or mismatched BH point indices')
    rows = [{'index': i, 'temperature_degC': arrays['Temperature'][i],
             'B_T': arrays['BValue_Magnet'][i], 'H_A_per_m': arrays['HValue_Magnet'][i]}
            for i in sorted(indices)]
    result = {'schema_version': '0.1', 'name': name,
              'source_file': source.name, 'source_sha256': hashlib.sha256(data).hexdigest(),
              'status': 'exported_structurally_checked_not_independently_validated',
              'raw_fields': fields, 'bh_points': rows,
              'warnings': ['No extrapolation or fitting performed.',
                           'Derived scalar parameters retain Motor-CAD reference temperature.',
                           'Source GDB curve fidelity has not been independently checked.']}
    target = source.with_suffix('.json')
    target.write_text(json.dumps(result, ensure_ascii=False, indent=2), encoding='utf-8')
    with source.with_suffix('.csv').open('w', newline='', encoding='utf-8-sig') as stream:
        writer = csv.DictWriter(stream, fieldnames=list(rows[0]))
        writer.writeheader()
        writer.writerows(rows)
    loaded = json.loads(target.read_text(encoding='utf-8'))
    assert loaded['raw_fields'] == fields
    assert loaded['bh_points'] == rows
    with source.with_suffix('.csv').open(encoding='utf-8-sig', newline='') as stream:

    print(json.dumps({'material': name, 'point_counts_by_temperature_degC': dict(counts),
                      'json_csv_roundtrip': 'passed'}, ensure_ascii=False, indent=2))
