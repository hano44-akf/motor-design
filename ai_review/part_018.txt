def switching_segments(branch_V,bus_V):
    branch=np.asarray(branch_V,dtype=float)
    if branch.shape!=(3,) or not np.all(np.isfinite(branch)) or not math.isfinite(bus_V) or bus_V<=0:
        raise ValueError('Three finite Delta voltages and positive bus required')
    if abs(branch.sum())>1e-9*max(1,float(abs(branch).max())):raise ValueError('Delta KVL mismatch')
    poles=np.array([branch[0],0.,-branch[1]])
    poles-=(poles.max()+poles.min())/2
    duty=.5+poles/bus_V
    if duty.min()<0 or duty.max()>1:raise ValueError('Requested voltage outside ideal inverter range')
    times=sorted(set([0.,1.]+list(duty/2)+list(1-duty/2)));segments=[]
    for lo,hi in zip(times[:-1],times[1:]):
        if hi<=lo:continue
        mid=(lo+hi)/2;carrier=2*mid if mid<=.5 else 2*(1-mid)
        state=(duty>carrier).astype(float)
        actual=bus_V*(state-np.roll(state,-1))
        segments.append((hi-lo,actual-branch))
    return duty,segments


def periodic_ripple(L_H,R_ohm,bus_V,branch_V,fsw_Hz,samples_per_segment=12):
    L=np.asarray(L_H,dtype=float)
    if L.shape!=(3,3) or not np.all(np.isfinite(L)) or not math.isfinite(R_ohm) or R_ohm<=0 or fsw_Hz<=0:
        raise ValueError('Finite 3x3 inductance, positive phase resistance and frequency required')
    if np.linalg.eigvalsh((L+L.T)/2).min()<=0:raise ValueError('Nonpositive symmetric energy part; no clipping applied')
    if not isinstance(samples_per_segment,int) or samples_per_segment<2:raise ValueError('At least two samples per segment')
    duty,segments=switching_segments(branch_V,bus_V);A=-R_ohm*np.linalg.inv(L);Ts=1/fsw_Hz
    c=np.zeros(3);transforms=[]
    for duration,delta_v in segments:
        dt=duration*Ts;F=expm(A*dt);offset=(np.eye(3)-F)@(delta_v/R_ohm)
        c=F@c+offset;transforms.append((dt,F,offset,delta_v))
    start=np.linalg.solve(np.eye(3)-expm(A*Ts),c);current=start.copy();samples=[]
    q,w=np.polynomial.legendre.leggauss(8);mean=np.zeros(3);second=np.zeros(3)
    for dt,F,offset,delta_v in transforms:
        steady=delta_v/R_ohm
        for t in np.linspace(0,dt,samples_per_segment+1):
            samples.append(steady+expm(A*t)@(current-steady))
        for t,weight in zip((q+1)*dt/2,w*dt/2/Ts):
            point=steady+expm(A*t)@(current-steady);mean+=weight*point;second+=weight*point**2
        current=F@current+offset
    samples=np.asarray(samples)
    return {'phase_ripple_pp_A':np.ptp(samples,axis=0).tolist(),
            'phase_ripple_rms_A':np.sqrt(np.maximum(0,second-mean**2)).tolist(),
            'periodic_residual_A':float(abs(current-start).max()),'mean_ripple_A':mean.tolist(),
            'duty':duty.tolist(),'qualified':False}


def csv_number(path,key):
    for row in csv.reader(path.read_text(encoding='cp1252').splitlines(),delimiter=';'):

            return float(d),row[2],float(abs(quantum)/2)
    raise ValueError('CSV unit evidence missing: '+key)

"""Independent fixed-grid RK4 / direct triangle-carrier numerical check.

Imports neither the preview's switch event schedule nor its matrix exponential
solver. Shares the frozen RL inputs, hence cannot qualify physical assumptions.
"""

def simulate(case,steps=16384):
    if isinstance(steps,bool) or not isinstance(steps,int) or steps<128:raise ValueError('At least 128 fixed steps required')
    L=np.asarray(case['inductance_H'],float);R=case['phase_resistance_ohm'];frequency=case['switching_frequency_Hz']
    bus=case['bus_V'];reference=np.asarray(case['branch_reference_V'],float)
    if L.shape!=(3,3) or not np.all(np.isfinite(L)) or not np.all(np.isfinite([R,frequency,bus])) or R<=0 or frequency<=0 or bus<=0:
        raise ValueError('Finite RL circuit and positive frequency/bus required')
    if reference.shape!=(3,) or not np.all(np.isfinite(reference)):raise ValueError('Three finite Delta branch voltages required')
    if abs(reference.sum())>1e-9:raise ValueError('Delta KVL mismatch')
    # Independent pole reconstruction via the Delta incidence matrix.
    C=np.array([[1.,-1.,0.],[0.,1.,-1.],[-1.,0.,1.]])
    poles=np.linalg.lstsq(C,reference,rcond=None)[0]
    offset=-.5*(poles.max()+poles.min());duty=.5+(poles+offset)/bus
    if duty.min()<0 or duty.max()>1:raise ValueError('Unsupported inverter command')
    def forcing(t):
        carrier=1-np.abs(2*t-1)
        switches=(carrier[:,None]<duty[None,:]).astype(float)
        voltage=bus*switches@C.T-reference
        return np.linalg.solve(L,voltage.T).T,voltage
    t=np.arange(steps)/steps;u0,v0=forcing(t);um,vm=forcing(t+.5/steps);u1,v1=forcing(t+1/steps)
    h=1/(steps*frequency);A=np.linalg.solve(L,-R*np.eye(3));Z=A*h;I=np.eye(3)
    Z2=Z@Z;Z3=Z2@Z;P=I+Z+Z2/2+Z3/6+Z3@Z/24
    # Algebraically collected classical RK4 stages, no matrix exponentials.
    K0=h/6*(I+Z+Z2/2+Z3/4);Km=h/6*(4*I+2*Z+Z2/2);K1=h/6*I
    kicks=u0@K0.T+um@Km.T+u1@K1.T
    forced=np.zeros(3)
    for kick in kicks:forced=P@forced+kick
    transition=np.linalg.matrix_power(P,steps);start=np.linalg.solve(I-transition,forced)
    current=start.copy();wave=np.empty((steps+1,3));wave[0]=current
    for i,kick in enumerate(kicks):current=P@current+kick;wave[i+1]=current
    return {'steps_per_carrier':steps,'phase_ripple_pp_A':np.ptp(wave,axis=0).tolist(),
            'mean_ripple_A':np.trapezoid(wave,dx=1/steps,axis=0).tolist(),
            'carrier_mean_voltage_error_V':((v0+4*vm+v1).mean(axis=0)/6).tolist(),
            'periodic_residual_A':float(np.max(abs(current-start))),
            'qualified':False}

def same(a,b):
    # Floating-point serialization tolerance only, not a physical validity range.
    if isinstance(a,(int,float)) and not isinstance(a,bool) and isinstance(b,(int,float)) and not isinstance(b,bool):
        return math.isfinite(a) and math.isfinite(b) and math.isclose(a,b,rel_tol=1e-12,abs_tol=1e-12)
    if isinstance(a,dict) and isinstance(b,dict):return a.keys()==b.keys() and all(same(v,b[k]) for k,v in a.items())
    if isinstance(a,list) and isinstance(b,list):return len(a)==len(b) and all(same(x,y) for x,y in zip(a,b))
    return a==b
