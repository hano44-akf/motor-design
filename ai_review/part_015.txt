def constraints(points,d,diameter,gap,wall):

    rows=[];jac=[]
    for i in range(n):
        for value,gradient in [
            (x[i]-d['body_front_x']-r-wall,[1.,0.]),
            (d['back_R']-d['Liner_Thickness']-radius[i]-r-wall,-p[i]/max(radius[i],1e-15)),
            (x[i]*math.sin(d['beta'])-c*math.cos(d['beta'])+y[i]*math.cos(d['beta'])-r-wall,[math.sin(d['beta']),math.cos(d['beta'])]),
            (x[i]*math.sin(d['beta'])-c*math.cos(d['beta'])-y[i]*math.cos(d['beta'])-r-wall,[math.sin(d['beta']),-math.cos(d['beta'])])]:
            row=np.zeros((n,2));row[i]=gradient;rows.append(value);jac.append(row.ravel())
    for i in range(n):

def starts(d,target,seed,count):

    # starting points, not a discretisation of the optimisation variables.
    for k in range(count):

        if counts.sum()!=target:raise ValueError('Start row count mismatch')
        points=[]
        for row,amount in enumerate(counts):

            half=max(.05,(x*math.tan(d['beta'])-(d['Tooth_Width']/2+d['Liner_Thickness'])/math.cos(d['beta']))-.95)
            ys=np.linspace(-half,half,amount) if amount>1 else [0.]

        p=np.array(points)
        if k>=3:p+=rng.normal(0,.2+(k%4)*.2,p.shape)
        if k>=6 and k%3==0:
            # Break row topology with farthest-point starts in the cavity.
            candidates=np.c_[rng.uniform(front,back,400),rng.uniform(-6,6,400)]
            mask=(candidates[:,0]*math.tan(d['beta'])-(d['Tooth_Width']/2+d['Liner_Thickness'])/math.cos(d['beta'])-abs(candidates[:,1]))>.5

            for _ in range(target-1):

def solve(d,target,diameter,gap,wall,restarts=18,maxiter=300,seed=20261105):
    if isinstance(target,bool) or not isinstance(target,int) or target<2:raise ValueError('At least two wires required')
    if not all(math.isfinite(v) for v in (diameter,gap,wall)) or diameter<=0 or min(gap,wall)<0:
        raise ValueError('Positive finite diameter and nonnegative gaps required')
    if any(isinstance(v,bool) or not isinstance(v,int) or v<1 for v in (restarts,maxiter,seed)):
        raise ValueError('Positive integer budgets and seed required')
    started=time.perf_counter();history=[];best=None
    extent=d['back_R']*math.tan(d['beta'])
    bounds=[(d['body_front_x'],d['back_R']-d['Liner_Thickness']),(-extent,extent)]*target+[(-diameter,diameter)]
    def values(z):
        c,_=constraints(z[:-1].reshape(target,2),d,diameter,gap,wall);return c-z[-1]
    def jac(z):
        _,J=constraints(z[:-1].reshape(target,2),d,diameter,gap,wall);return np.c_[J,-np.ones(len(J))]
    objective_gradient=np.r_[np.zeros(2*target),-1.]
    for index,p in enumerate(starts(d,target,seed,restarts)):
        z=np.r_[p.ravel(),min(0,float(constraints(p,d,diameter,gap,wall)[0].min()))]
        result=minimize(lambda z:-z[-1],z,jac=lambda z:objective_gradient,method='SLSQP',bounds=bounds,
                        constraints={'type':'ineq','fun':values,'jac':jac},options={'maxiter':maxiter,'ftol':1e-11})
        points=result.x[:-1].reshape(target,2);c,_=constraints(points,d,diameter,gap,wall);slack=float(c.min())
        history.append({'start':index,'iterations':int(result.nit),'success':bool(result.success),'message':str(result.message),
                        'minimum_constraint_slack_mm':slack,'objective_calls':int(result.nfev)})
        if best is None or slack>best['minimum_constraint_slack_mm']:
            best={'points_radial_tangential_mm':points.tolist(),'minimum_constraint_slack_mm':slack,'start':index}
        if slack>1e-7:break
    return {'algorithm':'SciPy SLSQP multistart max-min physical clearance','target_wires':target,
        'free_coordinate_variables':2*target,'auxiliary_slack_variables':1,'seed':seed,
        'restart_budget':restarts,'iteration_budget_per_start':maxiter,'completed_starts':len(history),'history':history,
        'best':best,'solver_found_nonnegative_slack':best['minimum_constraint_slack_mm']>=0,
        'elapsed_s':time.perf_counter()-started,'failure_is_impossibility_proof':False}


def verify_native_boundary(points,region,beta,diameter,gap,wall):

                conditions.append(np.linalg.norm(point-centre)<=radius+1e-9)

def segment_distance(a,b,c,d):
    """Minimum distance between two polyline segment sets, in any dimension.

    Minimise the convex quadratic over [0,1]^2: four box edges plus the
    unconstrained stationary point when it lies inside. Handles degenerate ends.
    """
    a,b,c,d=map(lambda v:np.asarray(v,float),(a,b,c,d))
    if a.ndim!=2 or a.shape!=b.shape or c.ndim!=2 or c.shape!=d.shape or a.shape[1]!=c.shape[1] or not all(np.all(np.isfinite(v)) for v in (a,b,c,d)):
        raise ValueError('Aligned finite segment endpoint arrays required')
    u=(b-a)[:,None,:];v=(d-c)[None,:,:];w=a[:,None,:]-c[None,:,:]
    A=np.sum(u*u,axis=2);B=np.sum(u*v,axis=2);C=np.sum(v*v,axis=2)
    D=np.sum(u*w,axis=2);E=np.sum(v*w,axis=2)
    def distance(s,t):return np.sum((w+s[...,None]*u-t[...,None]*v)**2,axis=2)
    safeA=np.where(A>0,A,1);safeC=np.where(C>0,C,1)
    zero=np.zeros_like(B);one=np.ones_like(B)
    result=np.minimum.reduce([distance(zero,np.clip(E/safeC,0,1)),distance(one,np.clip((E+B)/safeC,0,1)),
                              distance(np.clip(-D/safeA,0,1),zero),distance(np.clip((B-D)/safeA,0,1),one)])
    determinant=A*C-B*B;safe=np.where(determinant>1e-14*np.maximum(A*C,1e-30),determinant,1)
    s=(B*E-C*D)/safe;t=(A*E-B*D)/safe
    valid=(determinant>1e-14*np.maximum(A*C,1e-30))&(s>=0)&(s<=1)&(t>=0)&(t<=1)
    return float(np.sqrt(np.maximum(0,np.min(np.where(valid,np.minimum(result,distance(s,t)),result)))))


def bezier(control,t,derivative=0):
    P=np.asarray(control,float);t=np.asarray(t,float).reshape(-1,1)
    if P.ndim!=2 or P.shape[1]!=3 or len(P)<3 or not np.all(np.isfinite(P)) or not np.all(np.isfinite(t)):raise ValueError('Finite 3D Bezier controls required')
    if derivative not in (0,1,2):raise ValueError('Derivative 0..2 required')
    for _ in range(derivative):P=(len(P)-1)*np.diff(P,axis=0)
    degree=len(P)-1
    weights=np.hstack([math.comb(degree,k)*(1-t)**(degree-k)*t**k for k in range(degree+1)])
    return weights@P


def arch(a,b,peak_height):

def colour_conflicts(edges,separation):
    neighbours=[set() for _ in edges]
    for i,e in enumerate(edges):
        for j in range(i):
            f=edges[j]
            if segment_distance([e['a']],[e['b']],[f['a']],[f['b']])<separation:
                neighbours[i].add(j);neighbours[j].add(i)
    colours={}
    while len(colours)<len(edges):
        remaining=set(range(len(edges)))-colours.keys()
        node=max(remaining,key=lambda i:(len({colours[j] for j in neighbours[i] if j in colours}),len(neighbours[i]),-i))
        used={colours[j] for j in neighbours[node] if j in colours};colour=0
        while colour in used:colour+=1
        colours[node]=colour
    return [colours[i] for i in range(len(edges))],sum(map(len,neighbours))//2


def curve_statistics(control,intervals):
    if isinstance(intervals,bool) or not isinstance(intervals,int) or intervals<4:raise ValueError('At least four curve intervals required')
    t=np.linspace(0,1,intervals+1);points=bezier(control,t)
    P=np.asarray(control,float);degree=len(P)-1
    # The derivative curve lies in the convex hull of its derivative controls.
    # Unlike an endpoint sample, this is valid for the higher-degree detours too.
    second=degree*(degree-1)*np.diff(P,n=2,axis=0)
    bound=float(np.linalg.norm(second,axis=1).max()/(8*intervals**2))
    q,w=np.polynomial.legendre.leggauss(48)
    length=float(np.sum(w*np.linalg.norm(bezier(control,(q+1)/2,1),axis=1))/2)
    d1=bezier(control,t,1);d2=bezier(control,t,2);speed=np.linalg.norm(d1,axis=1)
    curvature=np.linalg.norm(np.cross(d1,d2),axis=1)/speed**3
    return {'points':points,'linearisation_error_bound_mm':bound,'length_mm':length,
            'sampled_minimum_bend_radius_mm':float(1/curvature.max()) if curvature.max()>0 else None}


def check_curves(routes,separation,intervals):

            'unresolved_pairs':sum(p['state']=='unresolved_near_limit' for p in pairs),

def pair_self_minimum(plus,minus,phase):
    pg=plus['graphs'][f'FluxLinkageLoadPh{phase}'];mg=minus['graphs'][f'FluxLinkageLoadPh{phase}']
    pi=plus['graphs'][f'PhaseCurrent{phase}'];mi=minus['graphs'][f'PhaseCurrent{phase}']
    x=np.asarray(pg['x']);dflux=np.asarray(pg['y'])-np.interp(x,mg['x'],mg['y'])
    di=np.interp(x,pi['x'],pi['y'])-np.interp(x,mi['x'],mi['y'])
    if np.any(abs(di)<1e-8):raise ValueError('Phase perturbation lost')
    return float(np.min(dflux/di))
