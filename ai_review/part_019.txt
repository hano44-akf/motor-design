"""Budgeted lateral/height detours, with equal-resolution before/after gates.

Cubic degree elevation preserves the initial curve. Only inner quintic controls
move, so conductor endpoints and axial exit tangents remain fixed. Bounded
heuristic search is not a routing/manufacturing or global-optimum certificate.
"""

def elevate(control):
    P=np.asarray(control,float)
    if P.shape!=(4,3) or not np.all(np.isfinite(P)):raise ValueError('Finite cubic controls required')
    while len(P)<6:
        n=len(P)-1
        P=np.array([P[0]]+[(i/(n+1))*P[i-1]+(1-i/(n+1))*P[i] for i in range(1,n+1)]+[P[-1]])
    return P


def detour(base,lateral_mm,height_delta_mm):
    base=np.asarray(base,float)
    if base.shape!=(6,3) or not np.all(np.isfinite(base)) or not all(math.isfinite(v) for v in (lateral_mm,height_delta_mm)):
        raise ValueError('Finite quintic controls and offsets required')
    if abs(lateral_mm)>3 or abs(height_delta_mm)>4:raise ValueError('Detour outside bounded proposal domain')
    delta=base[-1,:2]-base[0,:2];length=np.linalg.norm(delta)
    if length<=0:raise ValueError('Distinct route endpoints required')
    normal=np.array([-delta[1],delta[0]])/length;P=base.copy()
    P[2:4,:2]+=normal*lateral_mm;P[2:4,2]+=height_delta_mm
    if P[:,2].min()<0:raise ValueError('Route would enter lamination half-space')
    return P


def pair_penalty(a,b,separation):
    error=a['linearisation_error_bound_mm']+b['linearisation_error_bound_mm']
    box=np.linalg.norm(np.maximum(0,np.maximum(a['points'].min(axis=0)-b['points'].max(axis=0),b['points'].min(axis=0)-a['points'].max(axis=0))))
    if box-error>=separation:return 0.
    distance=segment_distance(a['points'][:-1],a['points'][1:],b['points'][:-1],b['points'][1:])
    return max(0,separation-max(0,distance-error))**2

"""Localise residual route gaps. Witness locations are on polylines, not proofs of cause."""

def closest_witness(points_a,points_b):
    a,b=np.asarray(points_a,float),np.asarray(points_b,float)
    if any(p.ndim!=2 or p.shape[1]!=3 or len(p)<2 or not np.all(np.isfinite(p)) for p in (a,b)):
        raise ValueError('Finite 3-D polylines required')
    u=np.diff(a,axis=0)[:,None,:];v=np.diff(b,axis=0)[None,:,:];w=a[:-1,None,:]-b[None,:-1,:]
    A=np.sum(u*u,axis=2);B=np.sum(u*v,axis=2);C=np.sum(v*v,axis=2)
    D=np.sum(u*w,axis=2);E=np.sum(v*w,axis=2)
    sa=np.where(A>0,A,1);sc=np.where(C>0,C,1);zero=np.zeros_like(B);one=np.ones_like(B)
    det=A*C-B*B;valid_det=det>1e-14*np.maximum(A*C,1e-30);safe=np.where(valid_det,det,1)
    s=(B*E-C*D)/safe;t=(A*E-B*D)/safe
    candidates=[(zero,np.clip(E/sc,0,1),np.ones_like(B,dtype=bool)),
                (one,np.clip((E+B)/sc,0,1),np.ones_like(B,dtype=bool)),
                (np.clip(-D/sa,0,1),zero,np.ones_like(B,dtype=bool)),
                (np.clip((B-D)/sa,0,1),one,np.ones_like(B,dtype=bool)),
                (s,t,valid_det&(s>=0)&(s<=1)&(t>=0)&(t<=1))]
    best=None
    for s,t,valid in candidates:
        distance2=np.where(valid,np.sum((w+s[...,None]*u-t[...,None]*v)**2,axis=2),np.inf)
        i,j=np.unravel_index(np.argmin(distance2),distance2.shape);value=float(distance2[i,j])
        if best is None or value<best[0]:best=(value,i,j,float(s[i,j]),float(t[i,j]))
    value,i,j,s,t=best
    return {'distance_mm':float(np.sqrt(max(0,value))),
            'a_t':(i+s)/(len(a)-1),'b_t':(j+t)/(len(b)-1),
            'a_xyz_mm':(a[i]+s*(a[i+1]-a[i])).tolist(),
            'b_xyz_mm':(b[j]+t*(b[j+1]-b[j])).tolist()}


def classify(routes,check,separation):
    by_id={r['id']:r for r in routes}
    if len(by_id)!=len(routes):raise ValueError('Unique route identifiers required')
    stats={r['id']:curve_statistics(r['control_points_xyz_mm'],check['intervals_per_curve']) for r in routes}
    rows=[];groups=Counter();incidents=Counter();coils=Counter()
    for pair in check['nonclear_pairs']:
        a,b=(by_id[pair[k]] for k in ('route_a','route_b'))
        if a['end']!=b['end']:raise ValueError('Pair belongs to different physical ends')
        witness=closest_witness(stats[a['id']]['points'],stats[b['id']]['points'])
        if abs(witness['distance_mm']-pair['polyline_distance_mm'])>1e-8:raise ValueError('Stored clearance distance does not match route geometry')
        same_coil=a['coil_index']==b['coil_index'];same_phase=a['phase']==b['phase']
        relation='same_coil' if same_coil else 'different_coil_same_phase' if same_phase else 'different_phase'
        # Parameter proximity is a location label, not a physical exit-length criterion.
        near_a=min(witness['a_t'],1-witness['a_t'])<=.1
        near_b=min(witness['b_t'],1-witness['b_t'])<=.1
        region='both_near_endpoints' if near_a and near_b else 'one_near_endpoint' if near_a or near_b else 'both_interior'
        pa=np.asarray(a['control_points_xyz_mm'])[[0,-1]];pb=np.asarray(b['control_points_xyz_mm'])[[0,-1]]
        endpoint_distance=float(np.linalg.norm(pa[:,None,:]-pb[None,:,:],axis=2).min())
        endpoint_blocked=endpoint_distance<separation
        row={**pair,'end':a['end'],'relation':relation,'region':region,
             'coil_a':a['coil_index'],'coil_b':b['coil_index'],'phase_a':a['phase'],'phase_b':b['phase'],
             'fixed_endpoint_minimum_distance_mm':endpoint_distance,'fixed_endpoints_below_required_gap':endpoint_blocked,
             'polyline_witness':witness}
        rows.append(row);groups[(a['end'],relation,region,pair['state'])]+=1
        for r in (a,b):incidents[r['id']]+=1
        for coil in {a['coil_index'],b['coil_index']}:coils[coil]+=1
    grouped=[{'end':k[0],'relation':k[1],'region':k[2],'state':k[3],'pairs':v} for k,v in sorted(groups.items())]
    return {'pairs':rows,'groups':grouped,'pair_count':len(rows),
            'by_relation':dict(Counter(r['relation'] for r in rows)),
            'by_region':dict(Counter(r['region'] for r in rows)),
            'by_end':dict(Counter(r['end'] for r in rows)),
            'fixed_endpoint_blocked_pairs':sum(r['fixed_endpoints_below_required_gap'] for r in rows),
            'route_priority':[{'route_id':k,'incident_pairs':v} for k,v in incidents.most_common()],
            'coil_priority':[{'coil_index':k,'incident_pairs':v} for k,v in coils.most_common()]}

"""Resolve overlapping UI settings without dropping or rounding fixed values."""
import math

DEFAULTS=(('Slot_Opening',1.,3.,'mm'),('Magnet_Arc_[ED]',40.,100.,'electrical degree'),('MagnetReduction',0.,1.5,'mm'))


def search_space(snapshot=None):

        if not isinstance(setting,dict):raise ValueError(source+': dimension setting object required')
        mode=setting.get('mode','auto')
        if mode=='auto':return
        if mode not in ('fixed','range'):raise ValueError(source+': fixed/range mode required')

            raise ValueError(source+': fixed value must be on the 0.1 mm / 0.1 ED development grid')
        sources[field].append({'source':source,'mode':mode,'min':lo,'max':hi})
    geometry=s.get('geometry',{});modes=s.get('modes',{});ranges=s.get('ranges',{})

        add(field,{'mode':mode,'value':geometry.get(key),**(ranges.get(key,{}) if mode=='range' else {})},'geometry.'+key,factor)
    stator=s.get('statorParameters',{})
    if 'slot_opening_mm' in stator:add('Slot_Opening',stator['slot_opening_mm'],'stator.slot_opening_mm')
    mappings={'pole_arc_ed':'Magnet_Arc_[ED]','magnet_reduction_mm':'MagnetReduction'}

    drafts=s.get('rotorParameters',{}).get('surface_breadloaf',{})

        for key,field in mappings.items():
            if key in settings:add(field,settings[key],origin+'.'+key)
    variables=[];bounds=[];combinations=1

"""Bounded shaft-torque current solve inside the local surrogate domain.

Current interval is the intersection of the development convex hull and the
user limit at fixed geometry. No current-angle/full-current-grid sweep.

import math

def current_interval(model,snapshot,max_current):
    if snapshot['expected_native']['ShaftSpeed']!=model['reference_speed_rpm']:return None
    if not math.isfinite(max_current) or max_current<=0:raise ValueError('Positive current limit required')
    n=snapshot['expected_native']
    geom=(np.array([n[k] for k in ('Slot_Opening','Magnet_Arc_[ED]','MagnetReduction')])-CENTER[:3])/SCALE[:3]
    training=np.asarray(model['training_coordinates']);a=np.vstack([np.ones(len(training)),training[:,:3].T]);b=np.r_[1.,geom]
    low=linprog(training[:,3],A_eq=a,b_eq=b,bounds=(0,None),method='highs')
    high=linprog(-training[:,3],A_eq=a,b_eq=b,bounds=(0,None),method='highs')
    if not low.success or not high.success:return None
    lo=max(0.,float(low.fun*SCALE[3]+CENTER[3]));hi=min(max_current,float(-high.fun*SCALE[3]+CENTER[3]))
    if lo>hi+1e-8:return None
    if hi-lo>2e-5:lo+=1e-5;hi-=1e-5
    return (lo,max(lo,hi))


def solve(proxy,snapshot,target_shaft_torque_Nm,max_line_peak_A,ref,contract,model):

            cache[key]=(local['shaft_torque_Nm'],screens)
        return cache[key]
    def unverified(reason):

        return {**fallback,'raw_current_matching_status':fallback['status'],'status':'shaft_matching_unverified',
                'shaft_control_status':'unverified','shaft_control_reason':reason,'control_torque_basis':'electromagnetic_lower_bound_only',
                'target_shaft_torque_Nm':target_shaft_torque_Nm,'shaft_control_evaluations':len(cache)}
    if interval is None:return unverified('No calibrated current interval at this geometry/speed within user current limit')
    lo,hi=interval
    try:
        low_t=point(lo)[0];high_t=point(hi)[0]
        if low_t>=target_shaft_torque_Nm:
            chosen=lo;matched=True;method='minimum_domain_current_already_above_minimum_target'
        elif high_t>=target_shaft_torque_Nm:
            chosen=brentq(lambda i:point(i)[0]-target_shaft_torque_Nm,lo,hi,xtol=1e-7)
            matched=True;method='bounded_brent_root'
        else:
            candidates=[lo,hi]
            if hi-lo>1e-7:
                r=minimize_scalar(lambda i:-point(i)[0],bounds=(lo,hi),method='bounded',options={'maxiter':24,'xatol':1e-5})

                chosen=brentq(lambda i:point(i)[0]-target_shaft_torque_Nm,lo,peak,xtol=1e-7)
                matched=True;method='bounded_peak_then_brent'
            else:chosen=peak;matched=False;method='best_within_domain_and_current_limit'
        shaft,screens=point(chosen)
    except ValueError as exc:return unverified(str(exc))
    raw=at_current(proxy,snapshot['line_peak_A'],chosen)
    return {**raw,'status':'matched_shaft_in_local_model' if matched else 'shaft_target_not_reached_within_domain',
            'control_torque_basis':'shaft_local_model','shaft_control_status':'model_target_met' if matched else 'model_target_unmet',
            'target_shaft_torque_Nm':target_shaft_torque_Nm,'predicted_shaft_torque_Nm':shaft,

            'scope':'Local surrogate shaft target, not actual FEA target satisfaction. Ripple remains raw prototype.'}

def dimensions(values):
    required=('Slot_Number','Stator_Bore','Slot_Depth','Tooth_Width','Slot_Opening','Tooth_Tip_Depth','Tooth_Tip_Angle','Slot_Corner_Radius','Liner_Thickness')

    if values.get('Slot_Type')!='Parallel_Tooth' or d['Slot_Corner_Radius']!=0:raise ValueError('Zero-fillet parallel-tooth template only')
    if d['Slot_Number']<3 or not d['Slot_Number'].is_integer():raise ValueError('Integer slot count required')
    if any(d[k]<=0 for k in required if k not in ('Slot_Corner_Radius','Liner_Thickness')) or d['Liner_Thickness']<0:
        raise ValueError('Positive dimensions and nonnegative liner required')
    if not 0<d['Tooth_Tip_Angle']<90:raise ValueError('Nonzero supported taper angle required')
    beta=math.pi/d['Slot_Number'];R=d['Stator_Bore']/2;opening=d['Slot_Opening']
    if opening>=2*R:raise ValueError('Opening exceeds bore')
    throat=math.sqrt(R*R-(opening/2)**2)+d['Tooth_Tip_Depth']
    slope=2/math.tan(math.radians(d['Tooth_Tip_Angle']))

    if denominator<=0:raise ValueError('Taper does not intersect supported tooth walls')
    front=(slope*throat-opening-d['Tooth_Width']/math.cos(beta))/denominator
    back=R+d['Slot_Depth']

    for liner in (0,d['Liner_Thickness']):
        if wall_halfwidth(d,front,liner)<=0:raise ValueError('Liner closes slot')
    return d


def wall_halfwidth(d,x,liner=0):
    return x*math.tan(d['beta'])-(d['Tooth_Width']/2+liner)/math.cos(d['beta'])


def corner_x(d,liner=0):
    t=math.tan(d['beta']);c=(d['Tooth_Width']/2+liner)/math.cos(d['beta']);R=d['back_R']-liner
    return (t*c+math.sqrt((1+t*t)*R*R-c*c))/(1+t*t)


def body_width(d,x,liner=0):
    x=np.asarray(x,float);R=d['back_R']-liner
    cap=np.sqrt(np.maximum(0,R*R-x*x))
    return np.where((x>=d['body_front_x'])&(x<=R),2*np.minimum(wall_halfwidth(d,x,liner),cap),0.)


def area(d,liner=0):
    front=d['body_front_x'];R=d['back_R']-liner;corner=corner_x(d,liner)
    # Split the line/circular-cap junction for numerical integration.
    return sum(quad(lambda x:float(body_width(d,x,liner)),a,b,epsabs=1e-10,epsrel=1e-11)[0]
               for a,b in ((front,corner),(corner,R)))


def native_width(region,x,beta):
    def rotate(p):return np.array([p['x']*math.cos(beta)+p['y']*math.sin(beta),-p['x']*math.sin(beta)+p['y']*math.cos(beta)])
    hits=[]
    for e in region['entities']:
        a=rotate(e['start']);b=rotate(e['end'])
        if e['type']=='line':
            if abs(b[0]-a[0])<1e-12:continue
            t=(x-a[0])/(b[0]-a[0])
            if -1e-10<=t<=1+1e-10:hits.append(a[1]+t*(b[1]-a[1]))
        elif e['type']=='arc':
            c=rotate(e['centre']);r=abs(e['radius']);disc=r*r-(x-c[0])**2
            if disc<0:continue
            sign=1 if e['radius']>0 else -1
            start=math.atan2(a[1]-c[1],a[0]-c[0]);end=math.atan2(b[1]-c[1],b[0]-c[0]);sweep=sign*(end-start)%(2*math.pi)
            for y in (c[1]-math.sqrt(disc),c[1]+math.sqrt(disc)):
                angle=math.atan2(y-c[1],x-c[0])
                if (sign*(angle-start))%(2*math.pi)<=sweep+1e-10:hits.append(y)
        else:raise ValueError('Unsupported boundary entity')
    unique=[]
    for y in sorted(hits):
        if not unique or abs(y-unique[-1])>1e-8:unique.append(y)
    if len(unique)!=2:raise ValueError('Expected two convex boundary intersections')
    return unique[-1]-unique[0]

def detailed_slot(snapshot,d,scenario,quadrature=96):
    from scipy.integrate import cumulative_trapezoid
    if snapshot['slots']!=d['Slot_Number'] or any(not math.isclose(snapshot['geometry_mm'][k],d[k],rel_tol=0,abs_tol=1e-9)

        raise ValueError('Snapshot/detail geometry mismatch')
    if isinstance(quadrature,bool) or not isinstance(quadrature,int) or quadrature<4:raise ValueError('Quadrature at least four required')
    sides=[[] for _ in range(snapshot['slots'])]
    for coil in snapshot['coils']:
        vec=np.eye(3)[coil['phase']]*coil['turns']

    if snapshot['paths']!=1 or any(len(v)!=2 for v in sides):raise ValueError('Two sides/slot, one path required')
    if scenario not in ('mixed','listed_bottom_first','listed_top_first'):raise ValueError('Unknown layer scenario')
    liner=d['Liner_Thickness'];front=d['body_front_x'];back=d['back_R'];conductor_back=back-liner

    grid=np.linspace(front,conductor_back,8193);width=body_width(d,grid,liner)
    integral=np.r_[0,cumulative_trapezoid(width,grid)];total_area=integral[-1]
    fraction=(total_area-integral)/total_area
    split=float(np.interp(.5,fraction[::-1],grid[::-1]))
    cuts=sorted(set([front,split,corner_x(d),corner_x(d,liner),conductor_back,back]))
    q,w=np.polynomial.legendre.leggauss(quadrature);matrix=np.zeros((3,3));L=snapshot['geometry_mm']['Stator_Lam_Length']/1000
    for slot in sides:

        for low,high in zip(cuts[:-1],cuts[1:]):

            enclosed=f[:,None]*turns if scenario=='mixed' else np.clip(f[:,None]*2-np.arange(2),0,1)@vectors
            # dy/b is dimensionless: mm cancel, stack length remains SI.

            'qualified':False,'used_for_guidance':False}
