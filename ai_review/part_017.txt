def align_by_current(proxy_angle_ed,proxy_current,native_angle_ed,native_current):
    def coefficients(angle,current):
        t=np.radians(angle);design=np.c_[np.cos(t),np.sin(t)]
        c=np.linalg.lstsq(design,np.asarray(current),rcond=None)[0]
        amplitude=np.linalg.norm(c,axis=0)
        if np.min(amplitude)<=1e-9:raise ValueError('Nonzero three phase fundamental required')
        return c/amplitude
    c=coefficients(proxy_angle_ed,proxy_current);d=coefficients(native_angle_ed,native_current)
    delta=math.atan2(float(np.sum(c[1]*d[0]-c[0]*d[1])),float(np.sum(c*d)))
    rotation=np.array([[math.cos(delta),math.sin(delta)],[-math.sin(delta),math.cos(delta)]])
    error=float(np.max(abs(rotation@c-d)))
    if error>1e-8:raise ValueError('Current phase sequence cannot be matched by one angle-origin shift')
    return math.degrees(delta),error


def interpolate_cycles(angle_ed,matrices,target_ed,shift_ed):
    angle=np.asarray(angle_ed,float);matrices=np.asarray(matrices,float)
    if matrices.shape!=(len(angle),3,3) or not np.all(np.isfinite(matrices)) or angle[0]!=0 or np.any(np.diff(angle)<=0):
        raise ValueError('Finite uniformly sampled full mechanical-cycle matrices required')
    step=angle[1]-angle[0]
    if not np.allclose(np.diff(angle),step) or not math.isclose((angle[-1]+step)/360,round((angle[-1]+step)/360)):
        raise ValueError('Integer electrical cycles required')
    cycles=round((angle[-1]+step)/360);n=round(360/step)
    if n*cycles!=len(angle):raise ValueError('Electrical cycles not aligned to samples')
    target=(np.asarray(target_ed)+shift_ed)%360;out=[]
    for cycle in range(cycles):
        values=matrices[cycle*n:(cycle+1)*n]
        sampled=np.empty((len(target),3,3))
        for i in range(3):
            for j in range(3):sampled[:,i,j]=np.interp(target,np.r_[np.arange(n)*step,360],np.r_[values[:,i,j],values[0,i,j]])
        out.append(sampled)
    out=np.asarray(out)
    return out.mean(axis=0),float(np.ptp(out,axis=0).max())


def diagnostics(estimate,measured):
    error=estimate-measured
    # Compare the balanced-current subspace as well as individual entries.
    Q=np.array([[1/math.sqrt(2),1/math.sqrt(6)],[-1/math.sqrt(2),1/math.sqrt(6)],[0,-2/math.sqrt(6)]])
    balanced=np.einsum('ai,nab,bj->nij',Q,error,Q)
    return {'mean_self_H':float(np.diagonal(estimate,axis1=1,axis2=2).mean()),
            'common_mode_max_abs_H':float(abs(estimate.sum(axis=(1,2))/3).max()),
            'matrix_rmse_H':float(np.sqrt(np.mean(error**2))),
            'maximum_entry_error_H':float(abs(error).max()),
            'balanced_subspace_error_spectral_max_H':float(np.linalg.norm(balanced,ord=2,axis=(1,2)).max()),
            'minimum_symmetric_energy_eigenvalue_H':float(np.linalg.eigvalsh((estimate+estimate.transpose(0,2,1))/2).min())}

"""Unfitted geometry/winding leakage prototype for one-path three-phase coils.

Slot matrix: mu0*l*sum integral m(y)m(y).T/b(y) dy from magnetic energy.
Uniform conductors, infinite-permeability walls, idealised parallel tooth slots.

Physical layer ordering is unknown: scenarios are NOT uncertainty bounds.

def positive(value,name):
    if isinstance(value,bool) or not isinstance(value,(int,float)) or not math.isfinite(value) or value<=0:
        raise ValueError('Positive finite '+name+' required')


def hanselman(coils_per_phase,turns,mean_coil_pitch_m,coil_area_m2):
    for key,value in locals().copy().items():positive(value,key)
    argument=mean_coil_pitch_m*math.sqrt(math.pi)/(2*math.sqrt(coil_area_m2))
    if argument<=1:raise ValueError('Hanselman logarithm domain unsupported; no clipping')
    return coils_per_phase*MU0*mean_coil_pitch_m*turns**2/2*math.log(argument)


def slot_energy_matrix(slot_sides,length_m,height_m,width_bottom_m,width_top_m,opening_depth_m,opening_width_m,scenario='mixed',quadrature=32):
    for name,value in [('length',length_m),('height',height_m),('bottom width',width_bottom_m),('top width',width_top_m),('opening width',opening_width_m)]:positive(value,name)
    if not math.isfinite(opening_depth_m) or opening_depth_m<0:raise ValueError('Nonnegative opening depth required')
    if scenario not in ('mixed','listed_bottom_first','listed_top_first'):raise ValueError('Unknown layer scenario')
    if isinstance(quadrature,bool) or not isinstance(quadrature,int) or quadrature<2:raise ValueError('Quadrature at least two required')
    q,w=np.polynomial.legendre.leggauss(quadrature);result=np.zeros((3,3))
    for sides in slot_sides:
        sides=np.asarray(sides,float)
        if sides.ndim!=2 or sides.shape[1]!=3 or not len(sides) or not np.all(np.isfinite(sides)):
            raise ValueError('Finite phase vectors for every slot side required')
        if scenario=='listed_top_first':sides=sides[::-1]
        total=sides.sum(axis=0)
        # Split exactly at layer boundaries before Gaussian integration.
        edges=np.linspace(0,1,len(sides)+1) if scenario!='mixed' else np.array([0.,1.])
        for low,high in zip(edges[:-1],edges[1:]):
            fraction=low+(q+1)*(high-low)/2
            if scenario=='mixed':enclosed=fraction[:,None]*total
            else:
                filled=np.clip(fraction[:,None]*len(sides)-np.arange(len(sides))[None,:],0,1)
                enclosed=filled@sides
            width=width_bottom_m+(width_top_m-width_bottom_m)*fraction
            result+=np.einsum('n,ni,nj->ij',w*(high-low)*height_m/(2*width),enclosed,enclosed)
        result+=opening_depth_m/opening_width_m*np.outer(total,total)
    return MU0*length_m*result


def estimate(snapshot,liner_mm,wire_gap_mm,scenario='mixed'):

CENTER=np.array([2.,70.,.4,38.]);SCALE=np.array([1.,20.,.4,5.])
TARGETS=('Stator Tooth Flux Density (maximum)','Stator Tooth Tip Flux Density (maximum)',
    'Stator Back Iron Flux Density (maximum)','Rotor Back Iron Flux Density (maximum)',
    'Stator iron Loss [total] (on load)','Rotor iron Loss [total] (on load)',
    'Magnet Loss (on load)','Average torque (MsVw)')


def coordinates(row):return (np.array([row['features'][k] for k in FEATURES],dtype=float)-CENTER)/SCALE


def bases(row):
    return np.array([row['raw_tooth_section_B_T']]*2+[row['raw_yoke_section_B_T']]*2+[1.]*3+
                    [row['raw_mean_electromagnetic_torque_Nm']])


def design(x):return np.column_stack([np.ones(len(x)),x])


def coefficients(x,y,regularization):
    a=design(x);penalty=np.diag([0.]+[regularization]*4)
    return np.linalg.solve(a.T@a+penalty,a.T@y)


def fit(rows):
    # Filtering happens before coordinates, target transforms and CV.
    train=[r for r in rows if r.get('training_allowed') and r['features']['ShaftSpeed']==3500]
    if len(train)<8:raise ValueError('At least eight development points at 3500rpm required')
    x=np.array([coordinates(r) for r in train]);native=np.array([[r['native'][k] for k in TARGETS] for r in train],dtype=float)
    base=np.array([bases(r) for r in train])
    if not np.all(np.isfinite(native)) or np.any(native<=0) or np.any(base<=0):raise ValueError('Positive finite native metrics and raw bases required')
    y=np.log(native/base);scores=[]
    for regularization in (1e-4,.001,.01,.1,1.,10.):
        residual=[]
        for i in range(len(train)):
            keep=np.arange(len(train))!=i;c=coefficients(x[keep],y[keep],regularization)
            residual.append(design(x[i:i+1])@c-y[i:i+1])
        scores.append((float(np.mean(np.array(residual)**2)),regularization))
    cv,regularization=min(scores);c=coefficients(x,y,regularization)
    loo=[]
    for i in range(len(train)):
        keep=np.arange(len(train))!=i;estimate=base[i]*np.exp((design(x[i:i+1])@coefficients(x[keep],y[keep],regularization))[0])
        loo.append(estimate-native[i])
    return {'schema':'breadloaf-local-loss-B-ridge-v1','training_ids':[r['id'] for r in train],
        'excluded_ids':[r['id'] for r in rows if r not in train],
        'features':list(FEATURES),'targets':list(TARGETS),'coefficients':c.tolist(),
        'regularization':regularization,'training_only_loo_log_mse':cv,
        'observed_loo_max_abs_error':dict(zip(TARGETS,np.max(np.abs(loo),axis=0).tolist())),
        'training_coordinates':x.tolist(),'reference_speed_rpm':3500,'qualified':False,
        'domain_policy':'Four-dimensional convex hull of development coordinates. Outside hull is unverified.',
        'uncertainty_policy':'LOO residual maxima are empirical diagnostics, not guaranteed or statistical bounds.'}


def predict(model,row,allow_extrapolation=False):
    if row['features'].get('ShaftSpeed')!=model['reference_speed_rpm']:
        return {'available':False,'reason':'Local shape model is at 3500rpm; frequency transfer is not connected','qualified':False}
    x=coordinates(row)
    try:inside=bool(Delaunay(np.array(model['training_coordinates'])).find_simplex(x,tol=1e-8)>=0)
    except QhullError:return {'available':False,'reason':'Development domain is rank-deficient','qualified':False}
    if not inside and not allow_extrapolation:return {'available':False,'reason':'Outside local development convex hull','qualified':False}
    values=bases(row)*np.exp((design(x[None,:])@np.array(model['coefficients']))[0])
    return {'available':True,'inside_development_hull':inside,'values':dict(zip(TARGETS,values.tolist())),
            'qualified':False,'scope':'3500rpm local surrogate; extrapolation diagnostic only' if not inside else 'Local experimental estimate, FEA required'}


def operating_estimate(model,snapshot,line_peak_A,raw_mean_torque_Nm,tooth_B,yoke_B,copper_W):
    row={'features':{**snapshot['expected_native'],'PeakCurrent':line_peak_A},
         'raw_tooth_section_B_T':tooth_B,'raw_yoke_section_B_T':yoke_B,
         'raw_mean_electromagnetic_torque_Nm':raw_mean_torque_Nm}
    p=predict(model,row)
    if not p['available']:return p
    values=p['values'];omega=2*math.pi*snapshot['expected_native']['ShaftSpeed']/60
    other=sum(values[k] for k in TARGETS[4:7]);pem=values[TARGETS[7]]*omega
    output=pem-other;input_power=pem+copper_W
    return {**p,'other_losses_W':other,'shaft_torque_Nm':output/omega,
            'efficiency_pct':100*output/input_power if output>=0 and input_power>0 else None,
            'observed_loo_max_abs_error':model['observed_loo_max_abs_error'],
            'training_ids':model['training_ids'],'local_peak_B_estimate_T':max(values[k] for k in TARGETS[:4])}


def run(batch):
    batch=Path(batch).resolve();data=json.loads((batch/'paired_dataset.json').read_text(encoding='utf-8'));model=fit(data['rows'])
    comparisons=[]
    for row in data['rows']:
        if row['role']!='shape_holdout':continue

        eta=100*(pem-other)/(pem+copper)
        comparisons.append({'id':row['id'],'inside_development_hull':result['inside_development_hull'],
            'prediction':result['values'],'native':row['native'],'differences':differences,
            'stator_iron_relative_error_pct':100*differences[TARGETS[4]]/row['native'][TARGETS[4]],
            'predicted_efficiency_pct':eta,'efficiency_error_percentage_points':eta-row['native']['System Efficiency'],
            'predicted_shaft_torque_Nm':(pem-other)/omega,'qualified':False})
    report={'schema':'breadloaf-local-loss-B-review-v1','model':model,'heldout_comparisons':comparisons,
        'new_fea_calls':0,'remaining_required_first_batch':data['counts']['remaining_required_first_batch'],
        'additional_fea_scheduled':0,'qualification':'Not family-wide validated. Two frozen shape checks only.',
        'training_includes_holdouts':False}
    write_json(batch/'local_loss_B_model.json',model);write_json(batch/'local_loss_B_review.json',report)

"""Compare saved search predictions with independent FEA; never refit or run FEA."""

def compare(study,batch):
    study=Path(study).resolve();batch=Path(batch).resolve()
    load=lambda p:json.loads(p.read_text(encoding='utf-8'))
    search=load(study/'result.json');native=load(batch/'result.json');plan=load(batch/'plan.json')
    candidates={f['id']:f for f in search['finalists']}
    cases={c['id']:c for c in plan['cases']};rows=[]
    for outcome in native['outcomes']:
        if outcome['status']!='completed':continue
        name=outcome['id']
        if name not in candidates:raise ValueError('Native case missing from saved search')
        prediction_path=study/(name+'_high_resolution.json')
        saved=load(prediction_path);matched=saved['matched']
        snapshot_path=batch/(name+'_snapshot.json');snapshot=load(snapshot_path)
        source=load(study/(name+'_snapshot.json'))
        for suffix,key in (('_snapshot.json','snapshot_sha256'),('_proxy.json','prediction_sha256')):
            if hashlib.sha256((batch/(name+suffix)).read_bytes()).hexdigest()!=cases[name][key]:
                raise ValueError('Frozen batch evidence changed')
        for key,value in source['expected_native'].items():
            if key!='PeakCurrent' and snapshot['expected_native'].get(key)!=value:
                raise ValueError('Prediction and Native input mismatch: '+key)
        if not math.isclose(matched['line_peak_A'],snapshot['line_peak_A'],abs_tol=1e-8,rel_tol=0):
            raise ValueError('Prediction and Native current mismatch')
        local=matched.get('model_screens',{}).get('local_calibrated_estimate',{})
        v=outcome['values']
        predictions={'ripple_pct':matched['ripple_pct'],
                     'shaft_torque_Nm':matched.get('predicted_shaft_torque_Nm'),
                     'efficiency_pct':local.get('efficiency_pct'),
                     'electromagnetic_torque_Nm':local.get('values',{}).get('Average torque (MsVw)')}
        actual={'ripple_pct':outcome['actual_ripple_pct'],'shaft_torque_Nm':v.get('Shaft Torque'),
                'efficiency_pct':v.get('System Efficiency'),'electromagnetic_torque_Nm':v.get('Average torque (MsVw)')}
        metrics={k:{'predicted':p,'actual':actual[k],
                    'prediction_minus_actual':p-actual[k] if p is not None and actual[k] is not None else None}
                 for k,p in predictions.items()}
        rows.append({'id':name,'metrics':metrics,'line_peak_A':snapshot['line_peak_A'],
                     'prediction_sha256':hashlib.sha256(prediction_path.read_bytes()).hexdigest()})

            'source_study':str(study),'source_batch':str(batch),
