"""Budget-limited FEA shape review, not a globally trained shape surrogate."""
import json
import time
from dimension_grid import mm


def refine(candidates,request,folder,calculate,progress,shortfall):
    began=time.perf_counter();op=request['operating_point'];bus=min(op['voltages_V']);temp=request['temperatures_C'][0]
    def rank(run):
        a=run['assessment']
        return (not a['physical_screens_pass'],not a['known_targets_met'],shortfall(run,request),-run['values']['System Efficiency'])
    for c in candidates:
        baseline=json.loads((folder/(c['tune_label']+'.json')).read_text(encoding='utf-8'))
        best=baseline;history=[];g=baseline['geometry_mm']
        # Design-directed samples; not a 0.1 mm exhaustive sweep.
        # Reduce slot depth to thicken the yoke; expose tooth width and magnet arc.
        for index,(depth_factor,tooth_factor,arc) in enumerate(((.85,1.,120),(.85,1.05,105),(.75,1.,90))):
            progress('detailed_shape_review',model=c['id'],trial=index+1)
            geometry={**c['design'].get('geometry_mm',{}),'Slot_Depth':mm(g['Slot_Depth']*depth_factor),
                      'Tooth_Width':mm(g['Tooth_Width']*tooth_factor)}
            d={**c['design'],'geometry_mm':geometry,'magnet_arc_deg':arc,
               'input_peak_current_A':baseline['design']['input_peak_current_A']*120/arc}
            trials=[]
            for correction in range(2):
                run=calculate(f"{c['id']}_detail{index}_tune{correction}",d,bus,temp,60)
                trials.append(run);torque=run['values']['Shaft Torque']
                if not run['assessment']['physical_screens_pass'] or op['torque_Nm']<=torque<=op['torque_Nm']*1.02:break
                d={**d,'input_peak_current_A':d['input_peak_current_A']*min(1.3,max(.7,op['torque_Nm']*1.005/max(torque,.001)))}
            # Do not reward overproduction via an artificially larger ripple denominator.
            matched=[r for r in trials if abs(r['values']['Shaft Torque']/op['torque_Nm']-1)<=.02]
            chosen=min(matched or trials,key=rank)
            history.append({'label':chosen['label'],'design':chosen['design'],
                            'shortfall':shortfall(chosen,request),'assessment':chosen['assessment']})
            if rank(chosen)<rank(best):best=chosen
        c['pre_detail_design']=baseline['design'];c['pre_detail_label']=baseline['label']
        c['detailed_review_trials']=history;c['design']=best['design'];c['tune_label']=best['label']
        # Keep the cheap prediction visibly tied to its original geometry.
        c['prediction_scope']='Initial proportional geometry only; no recalibrated detailed-shape prediction'
    (folder/'detailed_review.json').write_text(json.dumps({'wall_s':time.perf_counter()-began,
        'method':'Three directed shape alternatives per initial candidate, up to two current corrections; budget-limited FEA review, not exhaustive shape optimization',
        'candidates':[{'id':c['id'],'selected':c['tune_label'],'trials':c['detailed_review_trials']} for c in candidates]},indent=2),encoding='utf-8')

def make_plan():

        if shape=='Breadloaf':factors.append(('magnet_reduction_mm',.4,.2))

           'core':'POSCO | 50PN470','dimensions':{'diameter_mm':85,'length_mm':20},

            group=r['spm']['parameters'] if key in ('pole_arc_ed','magnet_reduction_mm') else r['geometry_parameters']
            group[key]={'mode':'range','min':mm(center-delta),'max':mm(center+delta)}

            for sign in (-1,1):

            recipe=compile_recipe(r,point)
            cases.append({'id':shape.lower()+'_'+str(index+1).zfill(2),'role':role,
                          'point':point,'recipe':recipe,'state':'proposed_not_executed'})
        groups.append({'shape':shape,'factor_count':len(factors),'factors':factors,'request':r,
            'mapping_save_readback_checks':1,'training_fea':1+2*len(factors),'holdout_fea':3,

"""Export saved run/prediction JSON without launching Motor-CAD.

Usage: python validation/export_csv.py INPUT.json --output-dir exports
Produces UTF-8 BOM CSVs and a strict JSON bundle usable by the UI.
"""
import argparse
import csv
import json
import math
from pathlib import Path
from csv_metadata import metadata as quantity_metadata

def flatten(value, prefix=''):
    if isinstance(value, dict):
        for key, child in value.items():
            yield from flatten(child, f'{prefix}.{key}' if prefix else str(key))
    elif isinstance(value, list):
        for i, child in enumerate(value):
            yield from flatten(child, f'{prefix}[{i}]')
    else:
        yield prefix, value

def clean(value):
    if isinstance(value, dict):
        return {k:clean(v) for k,v in value.items()}
    if isinstance(value, list):
        return [clean(v) for v in value]
    if isinstance(value,float) and not math.isfinite(value):
        return str(value)  # Preserve nonfinite state explicitly, never replace with zero.
    return value

def cell(value):
    if value is None:
        return ''
    if isinstance(value,str) and value.lstrip().startswith(('=','+','-','@')):
        return "'"+value
    return value

def export(source, output):
    data=clean(json.loads(source.read_text(encoding='utf-8-sig')))
    if not isinstance(data,dict):
        raise ValueError('Expected a JSON object')
    output.mkdir(parents=True,exist_ok=True)
    groups={'inputs':data.get('design',data.get('inputs',{})),
            'results':data.get('predictions',data.get('results',{})),
            'motorcad':data.get('values',{})}
    metadata={k:v for k,v in data.items() if k not in ('design','inputs','predictions','results','values')}
    for group,content in groups.items():
        with (output/f'{source.stem}_{group}.csv').open('w',encoding='utf-8-sig',newline='') as f:
            writer=csv.writer(f)
            writer.writerow(['source_file','category','field','value','data_status','unit','basis','condition_id','evidence'])
            status='saved_MotorCAD_output' if group=='motorcad' and content else 'prediction' if group=='results' and 'predictions' in data else 'saved_input_or_metadata'
            for key,value in flatten(content):
                unit,basis=quantity_metadata(key)
                writer.writerow([cell(source.name),group,cell(key),cell(value),status,unit,basis,'see_field_path',status])
            for key,value in flatten(metadata,'metadata'):
                unit,basis=quantity_metadata(key)
                writer.writerow([cell(source.name),'metadata',cell(key),cell(value),'source_metadata',unit,basis,'see_field_path','source_metadata'])
    bundle={'schema':'motor-csv-export-v1','source_file':source.name,'data':data}
    (output/f'{source.stem}_bundle.json').write_text(json.dumps(bundle,ensure_ascii=False,indent=2,allow_nan=False),encoding='utf-8')

"""Exact line/arc self-intersection screen in native mm coordinates.

Necessary boundary checks only; not a mesh or manufacturing certification.
"""
import math
import numpy as np

TOL=1e-6


def point(p):return np.array([p['x'],p['y']],dtype=float)
def cross(a,b):return float(a[0]*b[1]-a[1]*b[0])


def on_arc(p,e):
    c=point(e['centre']);r=e['radius'];a=point(e['start'])-c;b=point(e['end'])-c;v=p-c
    sign=1 if r>0 else -1
    start=math.atan2(a[1],a[0]);end=math.atan2(b[1],b[0]);angle=math.atan2(v[1],v[0])
    return (sign*(angle-start))%(2*math.pi)<=(sign*(end-start))%(2*math.pi)+1e-8


def intersections(a,b):
    if a['type']=='arc' and b['type']=='line':return intersections(b,a)
    p=point(a['start']);q=point(a['end'])
    if a['type']=='line' and b['type']=='line':
        r=point(b['start']);s=point(b['end']);u=q-p;v=s-r;den=cross(u,v)
        if abs(den)<1e-12:
            if abs(cross(r-p,u))>TOL:return []
            uu=float(u@u)
            if uu<TOL*TOL:return [p]
            lo,hi=sorted((float((r-p)@u/uu),float((s-p)@u/uu)))
            low=max(0,lo);high=min(1,hi)
            if high<low-1e-9:return []
            return [p+u*low] if high-low<1e-8 else [p+u*low,p+u*high,p+u*((low+high)/2)]
        t=cross(r-p,v)/den;z=cross(r-p,u)/den
        return [p+t*u] if -1e-9<=t<=1+1e-9 and -1e-9<=z<=1+1e-9 else []
    if a['type']=='line' and b['type']=='arc':
        c=point(b['centre']);r=abs(b['radius']);u=q-p;d=p-c
        aa=float(u@u);bb=2*float(d@u);cc=float(d@d)-r*r
        disc=bb*bb-4*aa*cc
        if aa<1e-15 or disc<-1e-9:return []
        out=[]
        for t in ((-bb-math.sqrt(max(0,disc)))/(2*aa),(-bb+math.sqrt(max(0,disc)))/(2*aa)):
            x=p+t*u
            if -1e-9<=t<=1+1e-9 and on_arc(x,b) and not any(np.linalg.norm(x-y)<TOL for y in out):out.append(x)
        return out
    c=point(a['centre']);d=point(b['centre']);r=abs(a['radius']);s=abs(b['radius']);v=d-c;distance=float(np.linalg.norm(v))
    if distance<TOL:
        if abs(r-s)>TOL:return []
        # Coincident arcs: overlap anywhere other than their shared endpoints
        # is not a simple boundary.
        sign=1 if a['radius']>0 else -1
        start=math.atan2(*(p-c)[::-1]);end=math.atan2(*(q-c)[::-1]);sweep=sign*(end-start)%(2*math.pi)
        pts=[p,q,c+r*np.array([math.cos(start+sign*sweep/2),math.sin(start+sign*sweep/2)])]
        pts.extend([point(b['start']),point(b['end'])])
        return [x for x in pts if on_arc(x,a) and on_arc(x,b)]
    if distance>r+s+TOL or distance<abs(r-s)-TOL:return []
    along=(r*r-s*s+distance*distance)/(2*distance);height=math.sqrt(max(0,r*r-along*along));unit=v/distance
    mid=c+along*unit;normal=np.array([-unit[1],unit[0]])
    return [x for x in (mid+height*normal,mid-height*normal) if on_arc(x,a) and on_arc(x,b)]


def check_region(region):
    edges=region.get('entities',[]);issues=[]
    if len(edges)<3:return {'passed':False,'issues':[{'reason':'too_few_edges'}]}
    for i,e in enumerate(edges):
        if e['type'] not in ('line','arc'):return {'passed':False,'issues':[{'reason':'unsupported_entity'}]}
        points=[point(e['start']),point(e['end'])]+([point(e['centre'])] if e['type']=='arc' else [])
        if any(not np.all(np.isfinite(p)) for p in points):return {'passed':False,'issues':[{'reason':'nonfinite_boundary'}]}
        if e['type']=='arc' and (not math.isfinite(e['radius']) or abs(e['radius'])<TOL):return {'passed':False,'issues':[{'reason':'invalid_arc_radius'}]}
        if np.linalg.norm(point(e['end'])-point(edges[(i+1)%len(edges)]['start']))>TOL:issues.append({'reason':'open_boundary','edge':i})
        if np.linalg.norm(point(e['end'])-point(e['start']))<TOL:issues.append({'reason':'zero_length_edge','edge':i})
    for i,a in enumerate(edges):
        for j in range(i+1,len(edges)):
            b=edges[j];shared=[point(a[k]) for k in ('start','end') for l in ('start','end') if np.linalg.norm(point(a[k])-point(b[l]))<TOL]
            for x in intersections(a,b):
                if (j==i+1 or (i==0 and j==len(edges)-1)) and any(np.linalg.norm(x-s)<TOL for s in shared):continue
                issues.append({'reason':'self_intersection','edges':[i,j],'point_mm':x.tolist()})
    return {'passed':not issues,'issues':issues,'scope':'Boundary closure and self-intersection only; no mesh or manufacturing qualification'}
