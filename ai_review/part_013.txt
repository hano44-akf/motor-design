def read_cases(folder,cases,completed):
    data={};hashes={}
    for c in cases:
        if c['id'] not in completed:raise ValueError('Requested incomplete case')
        path=folder/(c['id']+'.json');raw=path.read_bytes();r=json.loads(raw)
        if r['doe_role']!=c['role'] or r['doe_point']!=c['point'] or r['native_recipe']!=c['recipe']:
            raise ValueError('Case provenance mismatch: '+c['id'])
        if r['values']!=completed[c['id']]['values']:raise ValueError('Run summary differs from source')
        for k,v in r['geometry_mm'].items():
            if abs(v-c['recipe']['native_variables'][k])>1e-6:raise ValueError('Geometry mismatch')
        data[c['id']]=r;hashes[str(path.resolve())]=hashlib.sha256(raw).hexdigest()
    return data,hashes

def interpolate(pairs,x):
    points=sorted(set(pairs))
    for xx,yy in points:
        if x==xx:return yy
    for (xa,ya),(xb,yb) in zip(points,points[1:]):
        if xa<x<xb:return ya+(yb-ya)*(x-xa)/(xb-xa)
    return None

def analyze(material):
    fields={e['key']:e['value'] for e in material['raw_entries']}
    arrays=material.get('indexed_arrays',{})
    h={x['index']:float(x['raw_value']) for x in arrays.get('HValue',[])}
    b={x['index']:float(x['raw_value']) for x in arrays.get('BValue',[])}
    points=[(h[i],b[i]) for i in sorted(h.keys()&b.keys())]
    finite=all(math.isfinite(hh) and math.isfinite(bb) for hh,bb in points)
    increasing=finite and len(points)>1 and all(hb>ha and bb>ba for (ha,ba),(hb,bb) in zip(points,points[1:]))
    sampled={}
    for target in [1.4,1.5,1.6,1.7,1.8,1.9,2.0,2.076]:
        value=interpolate([(bb,hh) for hh,bb in points],target) if increasing else None
        below=interpolate([(bb,hh) for hh,bb in points],target-.01) if increasing else None
        above=interpolate([(bb,hh) for hh,bb in points],target+.01) if increasing else None
        sampled[str(target)]={'H_A_m':value,'incremental_relative_permeability_0p02T_window':.02/(MU0*(above-below)) if above is not None and below is not None and above>below else None}
    try:thickness=float(fields.get('LaminationThickness','nan'))
    except ValueError:thickness=None
    freq=sorted({float(e['raw_value']) for e in arrays.get('Frequency',[])})
    bs=[float(e['raw_value']) for e in arrays.get('FluxDensity',[])]
    return {'name':material['name'],'id':material['id'],'point_count':len(points),
            'monotone_original_order':increasing,'thickness_mm':thickness,
            'BH_range_H_A_m':[points[0][0],points[-1][0]] if points else None,
            'BH_range_B_T':[points[0][1],points[-1][1]] if points else None,
            'B_at_2500_A_m_T':interpolate(points,2500) if increasing else None,
            'B_at_5000_A_m_T':interpolate(points,5000) if increasing else None,

"""Offline, version-specific ripple validation audit. Never starts Motor-CAD."""
import argparse
import hashlib

def waveform_summary(wave):
    a = np.asarray(wave, dtype=float)
    if a.ndim != 1 or len(a) < 8 or not np.all(np.isfinite(a)):
        raise ValueError('Finite full-cycle waveform required')
    spectrum = np.fft.rfft(a - a.mean()) / len(a)
    amplitude = 2 * np.abs(spectrum)
    if len(a) % 2 == 0:
        amplitude[-1] /= 2
    orders = sorted(range(1, len(amplitude)), key=lambda k: amplitude[k], reverse=True)[:5]
    return {**ripple(a), 'ac_rms_Nm': float(np.std(a)),
            'dominant_mechanical_orders': [
                {'order': int(k), 'peak_amplitude_Nm': float(amplitude[k])} for k in orders]}


def compare(predicted, actual):
    p = np.asarray(predicted, dtype=float)
    a = np.asarray(actual, dtype=float)
    if p.shape != a.shape:
        raise ValueError('Matching mechanical-cycle grids required')
    ps, ac = waveform_summary(p), waveform_summary(a)
    # Phase search is a diagnostic only: do not replace or refit predictions.
    pc, ca = p - p.mean(), a - a.mean()
    errors = [float(np.mean((np.roll(pc, k) - ca) ** 2)) for k in range(len(p))]
    shift = int(np.argmin(errors))
    return {'prediction': ps, 'native': ac,
            'mean_error_pct': float(100 * (p.mean() - a.mean()) / abs(a.mean())) if abs(a.mean()) > 1e-12 else None,
            'ripple_error_percentage_points': ps['ripple_pct'] - ac['ripple_pct'] if ps['ripple_pct'] is not None and ac['ripple_pct'] is not None else None,
            'peak_to_peak_underprediction_factor': ac['peak_to_peak_Nm'] / ps['peak_to_peak_Nm'] if ps['peak_to_peak_Nm'] > 1e-12 else None,
            'best_circular_shift_mechanical_deg': shift * 360 / len(p),
            'best_shift_ac_rmse_Nm': float(np.sqrt(min(errors))),
            'phase_search_used_for_acceptance': False}


def audit(folder):

    rows = []
    for outcome in result['outcomes']:
        if outcome['status'] != 'completed':
            continue
        name = outcome['id']
        proxy_path = folder / (name + '_proxy.json')
        proxy = json.loads(proxy_path.read_text(encoding='utf-8'))
        snapshot = json.loads((folder / (name + '_snapshot.json')).read_text(encoding='utf-8'))
        graph = outcome['graphs']['TorqueMsVw']
        x, y = np.asarray(graph['x']), np.asarray(graph['y'])

"""Reconstruct the observed standard Breadloaf template, in millimetres.

Derived from saved native line/arc coordinates. Checked against the 16 saved
4p12s templates; not a claim about other Motor-CAD templates or versions.
"""
import math
from magnet_boundary_check import check_region


def boundary(bore_mm, gap_mm, thickness_mm, pole_arc_ed, reduction_mm, poles=4, axis_rad=math.pi/4):
    values=(bore_mm,gap_mm,thickness_mm,pole_arc_ed,reduction_mm,axis_rad)
    if any(not math.isfinite(v) for v in values) or poles<2 or poles%2:
        raise ValueError('Finite dimensions and even positive pole count required')
    if min(bore_mm,gap_mm,thickness_mm)>0 and 0<pole_arc_ed<180 and 0<=reduction_mm<thickness_mm:
        pass
    else:raise ValueError('Invalid Breadloaf dimensions')
    radius=bore_mm/2-gap_mm
    if radius<=thickness_mm:raise ValueError('No rotor iron radius')
    half=math.radians(pole_arc_ed/poles)
    y=radius*math.sin(half);inner=radius-thickness_mm
    outer=radius*math.cos(half)-reduction_mm
    edge=outer-inner
    if edge<=1e-6:raise ValueError('Breadloaf arc crosses backing line: nonpositive edge thickness')
    center=(radius*radius-outer*outer-y*y)/(2*(radius-outer))
    arc_radius=radius-center
    if arc_radius<=0:raise ValueError('Unsupported arc orientation')
    def point(x,y):
        return {'x':x*math.cos(axis_rad)-y*math.sin(axis_rad),
                'y':x*math.sin(axis_rad)+y*math.cos(axis_rad)}
    a,b,c,d=[point(x,z) for x,z in ((inner,-y),(outer,-y),(outer,y),(inner,y))]
    region={'region_type':'Magnet','centroid':point(radius-thickness_mm/2,0),
            'entities':[{'type':'line','start':a,'end':b},
                        {'type':'arc','start':b,'end':c,'centre':point(center,0),'radius':arc_radius},
                        {'type':'line','start':c,'end':d},{'type':'line','start':d,'end':a}]}
    # centroid above supplies the symmetry axis only, not an area centroid.
    if not check_region(region)['passed']:raise ValueError('Constructed boundary intersection')
    return region,{'edge_thickness_along_magnet_axis_mm':edge,
                   'centroid_scope':'Axis marker only, not area centroid',
                   'native_geometry_qualified':False}
