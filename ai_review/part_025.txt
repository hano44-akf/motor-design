"""Current matching in the linear coenergy prototype; no current-angle sweep.

This is a development model, not a saturation-aware drive feasibility solver.
Targets are electromagnetic torque, not shaft torque. Voltage/loss checks remain
separate. Component scaling is valid only for the uncalibrated linear model.
"""
import math
import numpy as np
from spm_ripple import ripple


def at_current(proxy, reference_line_peak_A, line_peak_A):
    if proxy.get('calibration') != 'none':
        raise ValueError('Raw linear coenergy components required')
    if not math.isfinite(reference_line_peak_A) or reference_line_peak_A <= 0:
        raise ValueError('Positive reference line peak current required')
    if not math.isfinite(line_peak_A) or line_peak_A < 0:
        raise ValueError('Nonnegative line peak current required')
    parts = [np.asarray(proxy[k], dtype=float) for k in ('cogging_Nm', 'mutual_Nm', 'reluctance_Nm')]
    if any(a.ndim != 1 or a.shape != parts[0].shape or len(a) < 8 or not np.all(np.isfinite(a)) for a in parts):
        raise ValueError('Matching finite component arrays required')
    scale = line_peak_A / reference_line_peak_A
    wave = parts[0] + scale * parts[1] + scale ** 2 * parts[2]
    return {**ripple(wave), 'torque_Nm': wave.tolist(), 'line_peak_A': float(line_peak_A),
            'current_scale': scale, 'qualified': False, 'optimization_eligible': False,
            'model': 'linear-coenergy-current-scaling-v1',
            'voltage_verified': False, 'saturation_verified': False}


def match_torque(proxy, reference_line_peak_A, target_electromagnetic_torque_Nm, max_line_peak_A):
    if not math.isfinite(target_electromagnetic_torque_Nm) or target_electromagnetic_torque_Nm <= 0:
        raise ValueError('Positive electromagnetic torque target required')
    if not math.isfinite(max_line_peak_A) or max_line_peak_A <= 0:
        raise ValueError('Positive maximum line peak current required')
    at_current(proxy, reference_line_peak_A, 0)  # Validate components/conventions.
    a, b, c = (float(np.mean(proxy[k])) for k in ('reluctance_Nm', 'mutual_Nm', 'cogging_Nm'))
    c -= target_electromagnetic_torque_Nm
    if abs(a) < 1e-12:
        roots = [-c / b] if abs(b) > 1e-12 else []
    else:
        roots = [float(z.real) for z in np.roots([a, b, c]) if abs(z.imag) < 1e-9]
    allowed = [max(0., s) for s in roots if s >= -1e-10 and s * reference_line_peak_A <= max_line_peak_A + 1e-9]
    if not allowed:
        scales=[0.,max_line_peak_A/reference_line_peak_A]
        if a<0 and 0<-b/(2*a)<scales[-1]:scales.append(-b/(2*a))
        best=max((at_current(proxy,reference_line_peak_A,s*reference_line_peak_A) for s in scales),key=lambda r:r['mean_Nm'])
        return {**best,'status': 'target_not_reached_within_current_limit',
                'target_electromagnetic_torque_Nm': target_electromagnetic_torque_Nm,
                'torque_shortfall_Nm':max(0,target_electromagnetic_torque_Nm-best['mean_Nm']),
                'max_line_peak_A': max_line_peak_A, 'qualified': False, 'optimization_eligible': False}
    result = at_current(proxy, reference_line_peak_A, min(allowed) * reference_line_peak_A)
    return {**result, 'status': 'matched_in_linear_model',
            'target_electromagnetic_torque_Nm': target_electromagnetic_torque_Nm,
            'scope': 'Id=0, unchanged geometry and phase, fixed-temperature linear model only; not FEA validation.'}

"""Offline current-matched Breadloaf study using saved, geometry-valid profiles."""
import argparse
import json
import time
from pathlib import Path

from magnet_boundary_check import check_region
from spm_ripple import ripple
from spm_ripple_operating_point import at_current, match_torque
from study_io import write_json


def run(source, output, target=.5, max_current=40):
    start=time.perf_counter();source=Path(source).resolve();output=Path(output).resolve()
    if output.exists():raise ValueError('Fresh output folder required')
    output.mkdir(parents=True)
    prior=json.loads((source/'result.json').read_text(encoding='utf-8'))
    native={r['id']:r for r in prior['outcomes'] if r['status']=='completed'}
    rows=[]
    for file in sorted(source.glob('surface_breadloaf_*_snapshot.json')):
        s=json.loads(file.read_text(encoding='utf-8'))
        if not check_region(s['magnet_region'])['passed']:continue
        saved=json.loads((source/(s['id']+'_proxy.json')).read_text(encoding='utf-8'))
        # Stored component arrays are the original raw C/M/R, not the corrected
        # total waveform. Preserve the source, build a separate raw record.
        raw={**saved,'calibration':'none'}
        reference=at_current(raw,s['line_peak_A'],s['line_peak_A'])
        matched=match_torque(raw,s['line_peak_A'],target,max_current)
        row={'id':s['id'],'reference_line_peak_A':s['line_peak_A'],
             'raw_reference_ripple_pct':reference['ripple_pct'],
             'raw_reference_mean_Nm':reference['mean_Nm'],
             'matched':{k:v for k,v in matched.items() if k!='torque_Nm'},
             'independent_native_at_matched_current':False,'qualified':False}
        if s['id'] in native:
            row['existing_native_reference_ripple_pct']=native[s['id']]['actual_ripple_pct']
            row['existing_native_reference_mean_Nm']=native[s['id']]['values']['Average torque (MsVw)']
        if 'torque_Nm' in matched:write_json(output/(s['id']+'_matched_waveform.json'),matched)
        rows.append(row)
    feasible=[r for r in rows if r['matched']['status']=='matched_in_linear_model']
    ranked=sorted(feasible,key=lambda r:r['matched']['ripple_pct'])
    report={'schema':'breadloaf-current-matching-study-v1','source':str(source),
            'target_electromagnetic_torque_Nm':target,'max_line_peak_A':max_current,
            'geometry_candidates':len(rows),'current_angle_sweep_count':0,'new_fea_count':0,
            'elapsed_s':time.perf_counter()-start,'rows':rows,
            'ranked_ids':[r['id'] for r in ranked],
            'predicted_le20_count':sum(r['matched']['ripple_pct']<=20 for r in feasible),
            'production_optimizer_connected':False,'qualified':False,
            'limitations':['Development torque target only, not a new user specification or shaft torque claim.',
                           'Fixed geometry pool, not a completed geometry optimization.',
                           'Linear, Id=0 model. No saturation, voltage or actual-current FEA validation.',
                           'Existing FEA is at reference current only; matched-current predictions are unverified.',
                           'Flat ripple is deferred by user; no Flat work performed.']}
    write_json(output/'result.json',report)
    return report


if __name__=='__main__':
    p=argparse.ArgumentParser();p.add_argument('--source',type=Path,required=True);p.add_argument('--output',type=Path,required=True)
    p.add_argument('--target-electromagnetic-torque',type=float,default=.5);p.add_argument('--max-line-peak-current',type=float,default=40)
    a=p.parse_args();r=run(a.source,a.output,a.target_electromagnetic_torque,a.max_line_peak_current)
    print(json.dumps(r,indent=2))

def write_json(path,data):
    temp=path.with_suffix(path.suffix+'.tmp')
    temp.write_text(json.dumps(clean(data),ensure_ascii=False,indent=2,allow_nan=False),encoding='utf-8')
    temp.replace(path)

def export_job(folder,result):
    header=['model','source','field','value','status','unit','basis','condition_id','evidence']
    groups={'inputs':[['','request',k,v,'requested'] for k,v in flatten(result['request'])],
            'results':[[c['id'],'selection',k,v,result['status']] for c in result['results'] for k,v in flatten({k:v for k,v in c.items() if k!='runs'})],
            'motorcad':[[c['id'],r['label'],k,v,'saved_MotorCAD_output'] for c in result['results'] for r in c['runs'] for k,v in flatten(r)]}
    groups['results'] += [['','job_metadata',k,v,result['status']] for k,v in flatten({k:v for k,v in result.items() if k not in ('request','results')})]

            for row in rows:
                model,source,field,value,status=row
                unit,basis=metadata(field)
                evidence=evidence_for(field,name,result.get('stage'))
                condition=contexts.get((model,source),'not_applicable_or_see_field_path')

                w.writerow([cell(v) for v in row+[unit,basis,condition,evidence]])

@contextlib.contextmanager
def worker_lock(db_path):
    handle=Path(db_path).with_suffix('.worker.lock').open('a+b')
    try:
        if handle.tell()==0:handle.write(b'0');handle.flush()
        handle.seek(0)
        if os.name=='nt':
            import msvcrt
            try:msvcrt.locking(handle.fileno(),msvcrt.LK_NBLCK,1)
            except OSError as exc:raise ValueError('A task worker is already active for this store') from exc
        else:
            import fcntl
            try:fcntl.flock(handle,fcntl.LOCK_EX|fcntl.LOCK_NB)
            except OSError as exc:raise ValueError('A task worker is already active for this store') from exc
        yield
    finally:handle.close()


def create_plan(store,title,tasks):
    return store.create(title,[prepare_spec(spec) for spec in tasks])


def run(store,job,allow_native=False,resume=False,retry_ids=(),reviewed=False):
    with worker_lock(store.path):
        store.begin_worker(job)
        if resume:store.set_stop(job,False)
        try:
            for ident in retry_ids:store.retry(job,ident,reviewed)
        except BaseException:
            store.finish_worker(job);raise
        while True:
            task=store.claim(job,allow_native)
            if task is None:break
            attempt_dir=Path(task['output_dir']);output=attempt_dir/'output'
            envelope={'kind':task['kind'],'payload':task['payload'],'attempt':task['attempt'],'output_dir':str(output)}
            write_json(attempt_dir/'task_input.json',envelope)
            process=None;timed_out=False;code=None
            try:
                with (attempt_dir/'worker.log').open('ab') as log:
                    process=subprocess.Popen([sys.executable,str(Path(__file__).resolve()),'--child',str(attempt_dir/'task_input.json')],
                        cwd=ROOT,stdout=log,stderr=subprocess.STDOUT,creationflags=getattr(subprocess,'CREATE_NO_WINDOW',0))
                    store.child_started(job,task['id'],task['attempt'],process.pid)
                    started=time.monotonic();stop_signalled=False
                    while process.poll() is None:
                        if time.monotonic()-started>task['timeout_s']:
                            timed_out=True;process.terminate();process.wait(timeout=20);break
                        if task['kind'] in NATIVE and not stop_signalled and store.is_stopped(job):
                            # Both native conventions are supported. Existing
                            # scripts stop at the next solve boundary.
                            (attempt_dir/'cancel.request').touch()
                            output.mkdir(exist_ok=True);(output/'cancel.request').touch();stop_signalled=True
                        time.sleep(.2)
                    code=process.returncode
                if timed_out:raise RuntimeError('Task timeout; review partial output and any native instance before retry')
                if code:raise RuntimeError(f'Task child exited {code}; see {attempt_dir / "worker.log"}')
                receipt=json.loads((output/'receipt.json').read_text(encoding='utf-8'))
                store.finish(job,task['id'],task['attempt'],output=receipt)
            except BaseException as exc:
                if process is not None and process.poll() is None:
                    process.terminate()
                    with contextlib.suppress(Exception):process.wait(timeout=20)
                store.finish(job,task['id'],task['attempt'],error=str(exc))
                if task['kind'] in NATIVE or isinstance(exc,(KeyboardInterrupt,SystemExit)):break
        return store.finish_worker(job)


def launch(store,job,allow_native=False,resume=False,retry_ids=(),reviewed=False):
    snapshot=store.snapshot(job)
    if not snapshot['implementation_compatible']:raise ValueError('Implementation changed; register a new plan')
    for ident in retry_ids:
        task=next((t for t in snapshot['tasks'] if t['id']==ident),None)
        if not task or task['state']!='failed':raise ValueError('Select failed task IDs only')
        if task['reason']=='input_changed':raise ValueError('Input changed; register a new plan')
        if (task['kind'] in NATIVE or task['reason'] in ('interrupted_worker_review_required','completed_artifact_changed')) and not reviewed:
            raise ValueError('Review partial results/process before retry')
    # Detect an already running queue before reporting a detached start.
    with worker_lock(store.path):pass
    log_dir=store.path.parent/'worker_logs';log_dir.mkdir(exist_ok=True)
    log_path=log_dir/(job+'_'+str(time.time_ns())+'.log')
    command=[sys.executable,str(Path(__file__).resolve()),'--db',str(store.path),'--run',job]
    if allow_native:command.append('--allow-native')
    if resume:command.append('--resume')
    if reviewed:command.append('--reviewed')
    for task in retry_ids:command.extend(['--retry',task])
    with log_path.open('ab') as log:
        process=subprocess.Popen(command,cwd=ROOT,stdout=log,stderr=subprocess.STDOUT,creationflags=getattr(subprocess,'CREATE_NO_WINDOW',0))
    return {'id':job,'worker_pid':process.pid,'worker_log':str(log_path),'native_enabled':allow_native,
            'note':'Detached worker started; task status and errors are persisted. No completion wait.'}


def worker_active(store):
    try:
        with worker_lock(store.path):return False
    except ValueError:return True
