"""Explicit Motor-CAD v261 SPM inputs, checked by readback before solving.

BPMRotor order was read from the installed radial-flux rotor selector;
see automation/rotor_enum_evidence.json. Magnetization: official automation FAQ.
"""
import math
from rotor_catalog import catalog,resolve_rotor,validate_parameters,default_magnetisation

ROTOR_TYPES={r['input_value']:r['motorcad_rotor_value'] for r in catalog()['rotors'] if r['input_mapping_status']=='verified'}
MAGNETISATION={'Parallel':0,'Radial':1}
TEMPERATURE_KEYS=['Magnet_Temperature','StatorConductor_Temperature','StatorLam_Temperature','RotorLam_Temperature']

def variables(options):
    rotor=resolve_rotor(options)
    direction=options.get('magnetisation',default_magnetisation(options))
    validate_parameters(options)
    if any(s.get('mode','auto')!='auto' for s in options.get('parameters',{}).values()):
        raise ValueError('Detailed rotor parameter mapping is not connected; values cannot be silently ignored')
    if rotor['input_mapping_status']!='verified' or direction not in rotor['verified_magnetisation']:
        raise ValueError('Motor-CAD input mapping pending for '+rotor['label']+' / '+direction)
    settings={'BPMRotor':rotor['motorcad_rotor_value'],'Magnetization':MAGNETISATION[direction],
              'ShaftSpeed':options.get('speed_rpm',3500),
              'WindingConnection':{'delta':1,'star':0}[options.get('connection','delta')],
              'NumberStrandsHand':options.get('strands',1),'ParallelPaths':options.get('paths',1),
              'Magnet_Arc_[ED]':options.get('magnet_arc_deg',120)}
    if 'phase_advance_deg' in options:
        angle=options['phase_advance_deg']
        if isinstance(angle,bool) or not isinstance(angle,(int,float)) or not math.isfinite(angle) or not -89.9<=angle<=89.9:
            raise ValueError('Phase advance must be -89.9..89.9 electrical degrees')
        settings['PhaseAdvance']=angle
    settings.update({k:options.get('temperature_C',20) for k in TEMPERATURE_KEYS})
    for key,value in settings.items():
        if not isinstance(value,(int,float)) or isinstance(value,bool) or not math.isfinite(value):
            raise ValueError('Invalid numeric setting: '+key)
    return settings

def apply_checked(mc, options):
    expected=variables(options)
    for key,value in expected.items():
        mc.set_variable(key,value)
    actual={key:mc.get_variable(key) for key in expected}
    for key,value in expected.items():
        if not isinstance(actual[key],(int,float)) or not math.isfinite(actual[key]) or abs(actual[key]-value)>1e-7:
            raise RuntimeError(f'Motor-CAD readback mismatch {key}: requested={value}, actual={actual[key]}')
    return actual

def write_status(folder,data):
    path=folder/'motorcad_open_status.json'
    temp=path.with_suffix('.json.tmp')
    temp.write_text(json.dumps(data,ensure_ascii=False,indent=2),encoding='utf-8');temp.replace(path)


def selected_model(folder,model_id,label=None):
    folder=Path(folder).resolve()
    if folder.parent!=JOBS.resolve() or not re.fullmatch(r'\d{8}_\d{6}(?:_\d+)?',folder.name):raise ValueError('Invalid saved job directory')
    result=json.loads((folder/'result.json').read_text(encoding='utf-8'))
    candidate=next((c for c in result['results'] if c['id']==model_id),None)
    if candidate is None:raise ValueError('Model is not in the saved comparison')
    runs=candidate['runs']
    if label is None:
        op=result['request']['operating_point']
        rated=op.get('nominal_voltage_V',12 if 12 in op['voltages_V'] else op['voltages_V'][0])
        target_temp=result['request'].get('temperatures_C',[20])[0]
        run=min(runs,key=lambda r:(abs(r['input_options']['temperature_C']-target_temp),abs(r['design']['voltage_V']-rated)))
    else:
        run=next((r for r in runs if r['label']==label),None)
        if run is None:raise ValueError('Operating condition is not in the saved model')
    if not re.fullmatch(r'[A-Za-z0-9_-]+',run['label']):raise ValueError('Invalid saved model label')
    path=(folder/(run['label']+'.mot')).resolve()
    if path.parent!=folder or not path.is_file():raise ValueError('Saved .mot model is unavailable')
    return path,run


def queue_open(folder,model_id,label=None):
    folder=Path(folder).resolve();path,run=selected_model(folder,model_id,label)
    status={'status':'queued','model_id':model_id,'label':run['label'],'source_model':str(path),'requested_at':time.time()}
    write_status(folder,status)
    args=[sys.executable,str(Path(__file__).resolve()),'--job',folder.name,'--model',model_id,'--label',run['label']]
    with (folder/'motorcad_open.log').open('ab') as log:
        proc=subprocess.Popen(args,cwd=str(ROOT.parent),stdout=log,stderr=subprocess.STDOUT,
                              creationflags=getattr(subprocess,'CREATE_NO_WINDOW',0))
    return {**status,'launcher_pid':proc.pid}

        raise
    finally:lock.close()

"""Candidate-local current proposal from measured electromagnetic torque.

No ripple calibration and no claim of saturation-aware extrapolation. A new
current always requires native validation. Shaft and electromagnetic targets
must not be mixed in the observations.
"""
import math


def propose_current(observations, target_torque_Nm, max_line_peak_A):
    if not observations:raise ValueError('At least one measured point required')
    def valid(x):return isinstance(x,(float,int)) and not isinstance(x,bool) and math.isfinite(x)
    if not valid(target_torque_Nm) or target_torque_Nm<=0 or not valid(max_line_peak_A) or max_line_peak_A<=0:
        raise ValueError('Positive finite target and current limit required')
    for point in observations:
        if not valid(point['line_peak_A']) or point['line_peak_A']<=0 or not valid(point['torque_Nm']) or point['torque_Nm']<=0:
            raise ValueError('Positive finite native observations required')
    points=sorted(observations,key=lambda p:abs(p['torque_Nm']-target_torque_Nm))
    a=points[0]
    if abs(a['torque_Nm']-target_torque_Nm)<=1e-12:
        current=a['line_peak_A'];method='already_at_target'
    elif len(points)==1:
        current=a['line_peak_A']*target_torque_Nm/a['torque_Nm'];method='single_point_local_proportional'
    else:
        b=next((p for p in points[1:] if abs(p['line_peak_A']-a['line_peak_A'])>1e-9),None)
        if b is None:raise ValueError('Distinct current observations required for secant')
        slope=(a['torque_Nm']-b['torque_Nm'])/(a['line_peak_A']-b['line_peak_A'])
        if slope<=1e-12:raise ValueError('Nonincreasing native torque; cannot extrapolate current')
        current=a['line_peak_A']+(target_torque_Nm-a['torque_Nm'])/slope;method='two_point_local_secant'
    within=0<current<=max_line_peak_A
    return {'status':'proposed_unverified' if within else 'proposal_outside_current_limit',
            'proposed_line_peak_A':current,'target_torque_Nm':target_torque_Nm,
            'max_line_peak_A':max_line_peak_A,'method':method,'within_current_limit':within,
            'native_verification_required':method!='already_at_target','qualified':False,
            'scope':'Same candidate, control and temperature only. No global torque correction or ripple refit.'}

        if not m:continue

        'coils_per_phase':counts,'coil_span_slots':spans,'slot_coil_side_counts':occupancy,

        'fundamental_winding_factors':factors,'harmonic_winding_factors':harmonics,
        'balanced':balanced,'synthesized_layout_equivalence':match,
        'coils':coils,'scope':'Slot/turn phasor and topology comparison only. Phase permutation/offset is a convention conversion, not permission to change native current ordering. Layer insulation, physical routing and manufacturability remain separate.'}

def basis(points):
    x=np.asarray(points,dtype=float)
    return np.column_stack([np.ones(len(x)),x,x*x,x[:,0]*x[:,1]])
